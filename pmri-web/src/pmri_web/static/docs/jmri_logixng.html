 60    def _wire(self, trigger: TriggerBinding) -> None:
 61        handler = self._make_handler(trigger)
 62        for kind, system_name in trigger.watches:
 63            bean = self._lookup_bean(kind, system_name)
 64            if bean is None:
 65                self.ctx.log.warning(
 66                    "trigger %r: no %s named %r — skipping watch",
 67                    trigger.user_name or trigger.conditional.id, kind, system_name,
 68                )
 69                continue
 70            self._disconnects.append(bean.state_changed.connect(handler))
 71
 72    def _make_handler(self, trigger: TriggerBinding) -> Callable[[Any], Any]:
 73        async def handle(_payload: Any) -> None:
 74            if trigger.guard is not None:
 75                guard_result = await trigger.guard.evaluate(self.ctx)
 76                if not guard_result:
 77                    self.ctx.emit_trace(
 78                        f"trigger {trigger.user_name or trigger.conditional.id}: guard false"
 79                    )
 80                    return
 81            self.ctx.emit_trace(
 82                f"trigger {trigger.user_name or trigger.conditional.id}: firing"
 83            )
 84            await trigger.conditional.execute(self.ctx)
 85        return handle
 86
 87    def _lookup_bean(self, kind: BeanKind, system_name: str) -> Any:
 88        # Each manager has a ``.get(name)`` method but they're different
 89        # generic types; an Any-typed bag keeps the union friendly.
 90        managers: dict[str, Any] = {
 91            "sensor": self.ctx.connection.sensors,
 92            "turnout": self.ctx.connection.turnouts,
 93            "light": self.ctx.connection.lights,
 94            "memory": self.ctx.connection.memories,
 95            "block": self.ctx.connection.blocks,
 96            "reporter": self.ctx.connection.reporters,
 97        }
 98        manager = managers.get(kind)
 99        if manager is None:
100            return None
101        return manager.get(system_name)
