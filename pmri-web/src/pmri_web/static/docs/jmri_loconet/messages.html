jmri_loconet.messages

Typed LocoNet message dataclasses + encode/decode dispatch.

Each opcode JMRI's LnConstants declares maps to one frozen dataclass. decode() turns frame bytes into the matching dataclass; encode() does the reverse (returns frame bytes including checksum).

Unrecognized but well-formed frames decode to OpcUnknown so listeners can still observe them — JMRI exposes everything on the bus.

235def decode(frame: bytes) -> LocoNetMessage:
236    """Turn a complete, checksum-verified frame into a typed message.
237
238    Raises :class:`ValueError` if the checksum is invalid.
239    """
240    if not verify_checksum(frame):
241        raise ValueError("LocoNet frame has invalid checksum")
242    op = frame[0]
243
244    # 2-byte
245    if op == OPC_GPON:
246        return OpcGpOn()
247    if op == OPC_GPOFF:
248        return OpcGpOff()
249    if op == OPC_IDLE:
250        return OpcIdle()
251
252    # 4-byte: layout is [op, b1, b2, cs]; b1/b2 vary per opcode.
253    if op == OPC_SW_REQ:
254        return OpcSwReq(
255            address=_decode_switch_address(frame[1], frame[2]),
256            closed=bool(frame[2] & 0x20),
257            output_on=bool(frame[2] & 0x10),
258        )
259    if op == OPC_SW_REP:
260        return OpcSwRep(
261            address=_decode_switch_address(frame[1], frame[2]),
262            closed=bool(frame[2] & 0x20),
263            thrown=not bool(frame[2] & 0x20),
264        )
265    if op == OPC_INPUT_REP:
266        # JMRI's mapping:
267        #   pair = ((SN2 & 0x0F) << 7) | (SN1 & 0x7F)
268        #   sensor (1-based) = pair * 2 + ((SN2 & 0x20) >> 5) + 1
269        pair = ((frame[2] & 0x0F) << 7) | (frame[1] & 0x7F)
270        sensor_addr = (pair * 2) + ((frame[2] & 0x20) >> 5) + 1
271        active = bool(frame[2] & 0x10)
272        return OpcInputRep(address=sensor_addr, active=active)
273    if op == OPC_LONG_ACK:
274        return OpcLongAck(responding_opcode=frame[1] | 0x80, ack_code=frame[2])
275    if op == OPC_SLOT_STAT1:
276        return OpcSlotStat1(slot=frame[1], stat=frame[2])
277    if op == OPC_MOVE_SLOTS:
278        return OpcMoveSlots(src=frame[1], dest=frame[2])
279    if op == OPC_LINK_SLOTS:
280        return OpcLinkSlots(src=frame[1], dest=frame[2])
281    if op == OPC_UNLINK_SLOTS:
282        return OpcUnlinkSlots(src=frame[1], dest=frame[2])
283    if op == OPC_RQ_SL_DATA:
284        return OpcRqSlData(slot=frame[1])
285    if op == OPC_LOCO_ADR:
286        return OpcLocoAdr(address=(frame[1] << 7) | frame[2])
287    if op == OPC_LOCO_SPD:
288        return OpcLocoSpd(slot=frame[1], speed=frame[2])
289    if op == OPC_LOCO_DIRF:
290        return OpcLocoDirF(slot=frame[1], dirf=frame[2])
291    if op == OPC_LOCO_SND:
292        return OpcLocoSnd(slot=frame[1], snd=frame[2])
293
294    # Variable-length: 14-byte slot data frames.
295    if op == OPC_SL_RD_DATA and len(frame) == 14:
296        return OpcSlRdData(
297            slot=frame[2], stat=frame[3],
298            address=(frame[9] << 7) | frame[4],
299            speed=frame[5], dirf=frame[6], trk=frame[7],
300            ss2=frame[8], snd=frame[10],
301        )
302    if op == OPC_WR_SL_DATA and len(frame) == 14:
303        return OpcWrSlData(
304            slot=frame[2], stat=frame[3],
305            address=(frame[9] << 7) | frame[4],
306            speed=frame[5], dirf=frame[6], trk=frame[7],
307            ss2=frame[8], snd=frame[10],
308        )
309
310    return OpcUnknown(opcode=op, payload=bytes(frame[1:-1]))

316def encode(msg: LocoNetMessage) -> bytes:
317    """Turn *msg* into a complete frame (opcode + payload + checksum)."""
318    op = msg.opcode
319
320    # 2-byte
321    if op in (OPC_GPON, OPC_GPOFF, OPC_IDLE):
322        return append_checksum(bytes([op]))
323
324    # 4-byte — isinstance narrows the type cleanly per branch.
325    if isinstance(msg, OpcSwReq):
326        b1, b2 = _encode_switch_address(
327            msg.address, closed=msg.closed, output_on=msg.output_on
328        )
329        return append_checksum(bytes([op, b1, b2]))
330    if isinstance(msg, OpcSwRep):
331        b1, b2 = _encode_switch_address(
332            msg.address, closed=msg.closed, output_on=False, report_mode=True
333        )
334        return append_checksum(bytes([op, b1, b2]))
335    if isinstance(msg, OpcInputRep):
336        addr = msg.address - 1
337        sensor_bit = addr & 0x01
338        addr_words = addr >> 1
339        b1 = addr_words & 0x7F
340        b2 = ((addr_words >> 7) & 0x0F) | (sensor_bit << 5) | (0x10 if msg.active else 0)
341        return append_checksum(bytes([op, b1, b2]))
342    if isinstance(msg, OpcLongAck):
343        return append_checksum(
344            bytes([op, msg.responding_opcode & 0x7F, msg.ack_code & 0x7F])
345        )
346    if isinstance(msg, OpcSlotStat1):
347        return append_checksum(bytes([op, msg.slot & 0x7F, msg.stat & 0x7F]))
348    if isinstance(msg, OpcMoveSlots):
349        return append_checksum(bytes([op, msg.src & 0x7F, msg.dest & 0x7F]))
350    if isinstance(msg, OpcLinkSlots):
351        return append_checksum(bytes([op, msg.src & 0x7F, msg.dest & 0x7F]))
352    if isinstance(msg, OpcUnlinkSlots):
353        return append_checksum(bytes([op, msg.src & 0x7F, msg.dest & 0x7F]))
354    if isinstance(msg, OpcRqSlData):
355        return append_checksum(bytes([op, msg.slot & 0x7F, 0x00]))
356    if isinstance(msg, OpcLocoAdr):
357        return append_checksum(
358            bytes([op, (msg.address >> 7) & 0x7F, msg.address & 0x7F])
359        )
360    if isinstance(msg, OpcLocoSpd):
361        return append_checksum(bytes([op, msg.slot & 0x7F, msg.speed & 0x7F]))
362    if isinstance(msg, OpcLocoDirF):
363        return append_checksum(bytes([op, msg.slot & 0x7F, msg.dirf & 0x7F]))
364    if isinstance(msg, OpcLocoSnd):
365        return append_checksum(bytes([op, msg.slot & 0x7F, msg.snd & 0x7F]))
366
367    # 14-byte slot data frames.
368    if isinstance(msg, (OpcSlRdData, OpcWrSlData)):
369        body = bytes([
370            op,
371            14,                  # length byte
372            msg.slot & 0x7F,
373            msg.stat & 0x7F,
374            msg.address & 0x7F,    # ADR (low)
375            msg.speed & 0x7F,
376            msg.dirf & 0x7F,
377            msg.trk & 0x7F,
378            msg.ss2 & 0x7F,      # SS2 — hours for slot 123, STAT2 elsewhere
379            (msg.address >> 7) & 0x7F,  # ADR2 (high)
380            msg.snd & 0x7F,
381            0x00,                # ID1
382            0x00,                # ID2
383        ])
384        return append_checksum(body)
385
386    if isinstance(msg, OpcUnknown):
387        body = bytes([msg.opcode]) + msg.payload
388        return append_checksum(body)
389
390    raise NotImplementedError(f"encode not implemented for opcode 0x{op:02X}")

396def _decode_switch_address(b1: int, b2: int) -> int:
397    """Decode a LocoNet switch (turnout) address from the two body bytes."""
398    # b1 = lower 7 bits, b2 low 4 bits = upper 4 bits; address is 1-based.
399    return ((b2 & 0x0F) << 7 | (b1 & 0x7F)) + 1
400
401
402def _encode_switch_address(
403    address: int, *, closed: bool, output_on: bool, report_mode: bool = False
404) -> tuple[int, int]:
405    """Encode a switch address into the two body bytes.
406
407    *report_mode* is True for OPC_SW_REP frames (bit-3 layout differs slightly
408    in JMRI; the visible difference is which bits carry feedback flags).
409    """
410    addr0 = address - 1
411    b1 = addr0 & 0x7F
412    b2 = (addr0 >> 7) & 0x0F
413    if closed:
414        b2 |= 0x20
415    if output_on and not report_mode:
416        b2 |= 0x10
417    return b1, b2

20def compute_checksum(body: bytes) -> int:
21    """Return the LocoNet checksum byte for *body* (everything before the CS).

23    The checksum is ``0xFF XOR`` of every byte in *body*.

def decode(frame: bytes) -> LocoNetMessage: 
236def decode(frame: bytes) -> LocoNetMessage:
237    """Turn a complete, checksum-verified frame into a typed message.
238
239    Raises :class:`ValueError` if the checksum is invalid.
240    """
241    if not verify_checksum(frame):
242        raise ValueError("LocoNet frame has invalid checksum")
243    op = frame[0]
244
245    # 2-byte
246    if op == OPC_GPON:
247        return OpcGpOn()
248    if op == OPC_GPOFF:
249        return OpcGpOff()
250    if op == OPC_IDLE:
251        return OpcIdle()
252
253    # 4-byte: layout is [op, b1, b2, cs]; b1/b2 vary per opcode.
254    if op == OPC_SW_REQ:
255        return OpcSwReq(
256            address=_decode_switch_address(frame[1], frame[2]),
257            closed=bool(frame[2] & 0x20),
258            output_on=bool(frame[2] & 0x10),
259        )
260    if op == OPC_SW_REP:
261        return OpcSwRep(
262            address=_decode_switch_address(frame[1], frame[2]),
263            closed=bool(frame[2] & 0x20),
264            thrown=not bool(frame[2] & 0x20),
265        )
266    if op == OPC_INPUT_REP:
267        # JMRI's mapping:
268        #   pair = ((SN2 & 0x0F) << 7) | (SN1 & 0x7F)
269        #   sensor (1-based) = pair * 2 + ((SN2 & 0x20) >> 5) + 1
270        pair = ((frame[2] & 0x0F) << 7) | (frame[1] & 0x7F)
271        sensor_addr = (pair * 2) + ((frame[2] & 0x20) >> 5) + 1
272        active = bool(frame[2] & 0x10)
273        return OpcInputRep(address=sensor_addr, active=active)
274    if op == OPC_LONG_ACK:
275        return OpcLongAck(responding_opcode=frame[1] | 0x80, ack_code=frame[2])
276    if op == OPC_SLOT_STAT1:
277        return OpcSlotStat1(slot=frame[1], stat=frame[2])
278    if op == OPC_MOVE_SLOTS:
279        return OpcMoveSlots(src=frame[1], dest=frame[2])
280    if op == OPC_LINK_SLOTS:
281        return OpcLinkSlots(src=frame[1], dest=frame[2])
282    if op == OPC_UNLINK_SLOTS:
283        return OpcUnlinkSlots(src=frame[1], dest=frame[2])
284    if op == OPC_RQ_SL_DATA:
285        return OpcRqSlData(slot=frame[1])
286    if op == OPC_LOCO_ADR:
287        return OpcLocoAdr(address=(frame[1] << 7) | frame[2])
288    if op == OPC_LOCO_SPD:
289        return OpcLocoSpd(slot=frame[1], speed=frame[2])
290    if op == OPC_LOCO_DIRF:
291        return OpcLocoDirF(slot=frame[1], dirf=frame[2])
292    if op == OPC_LOCO_SND:
293        return OpcLocoSnd(slot=frame[1], snd=frame[2])
294
295    # Variable-length: 14-byte slot data frames.
296    if op == OPC_SL_RD_DATA and len(frame) == 14:
297        return OpcSlRdData(
298            slot=frame[2], stat=frame[3],
299            address=(frame[9] << 7) | frame[4],
300            speed=frame[5], dirf=frame[6], trk=frame[7],
301            ss2=frame[8], snd=frame[10],
302        )
303    if op == OPC_WR_SL_DATA and len(frame) == 14:
304        return OpcWrSlData(
305            slot=frame[2], stat=frame[3],
306            address=(frame[9] << 7) | frame[4],
307            speed=frame[5], dirf=frame[6], trk=frame[7],
308            ss2=frame[8], snd=frame[10],
309        )
310
311    return OpcUnknown(opcode=op, payload=bytes(frame[1:-1]))

317def encode(msg: LocoNetMessage) -> bytes:
318    """Turn *msg* into a complete frame (opcode + payload + checksum)."""
319    op = msg.opcode
320
321    # 2-byte
322    if op in (OPC_GPON, OPC_GPOFF, OPC_IDLE):
323        return append_checksum(bytes([op]))
324
325    # 4-byte — isinstance narrows the type cleanly per branch.
326    if isinstance(msg, OpcSwReq):
327        b1, b2 = _encode_switch_address(
328            msg.address, closed=msg.closed, output_on=msg.output_on
329        )
330        return append_checksum(bytes([op, b1, b2]))
331    if isinstance(msg, OpcSwRep):
332        b1, b2 = _encode_switch_address(
333            msg.address, closed=msg.closed, output_on=False, report_mode=True
334        )
335        return append_checksum(bytes([op, b1, b2]))
336    if isinstance(msg, OpcInputRep):
337        addr = msg.address - 1
338        sensor_bit = addr & 0x01
339        addr_words = addr >> 1
340        b1 = addr_words & 0x7F
341        b2 = ((addr_words >> 7) & 0x0F) | (sensor_bit << 5) | (0x10 if msg.active else 0)
342        return append_checksum(bytes([op, b1, b2]))
343    if isinstance(msg, OpcLongAck):
344        return append_checksum(
345            bytes([op, msg.responding_opcode & 0x7F, msg.ack_code & 0x7F])
346        )
347    if isinstance(msg, OpcSlotStat1):
348        return append_checksum(bytes([op, msg.slot & 0x7F, msg.stat & 0x7F]))
349    if isinstance(msg, OpcMoveSlots):
350        return append_checksum(bytes([op, msg.src & 0x7F, msg.dest & 0x7F]))
351    if isinstance(msg, OpcLinkSlots):
352        return append_checksum(bytes([op, msg.src & 0x7F, msg.dest & 0x7F]))
353    if isinstance(msg, OpcUnlinkSlots):
354        return append_checksum(bytes([op, msg.src & 0x7F, msg.dest & 0x7F]))
355    if isinstance(msg, OpcRqSlData):
356        return append_checksum(bytes([op, msg.slot & 0x7F, 0x00]))
357    if isinstance(msg, OpcLocoAdr):
358        return append_checksum(
359            bytes([op, (msg.address >> 7) & 0x7F, msg.address & 0x7F])
360        )
361    if isinstance(msg, OpcLocoSpd):
362        return append_checksum(bytes([op, msg.slot & 0x7F, msg.speed & 0x7F]))
363    if isinstance(msg, OpcLocoDirF):
364        return append_checksum(bytes([op, msg.slot & 0x7F, msg.dirf & 0x7F]))
365    if isinstance(msg, OpcLocoSnd):
366        return append_checksum(bytes([op, msg.slot & 0x7F, msg.snd & 0x7F]))
367
368    # 14-byte slot data frames.
369    if isinstance(msg, (OpcSlRdData, OpcWrSlData)):
370        body = bytes([
371            op,
372            14,                  # length byte
373            msg.slot & 0x7F,
374            msg.stat & 0x7F,
375            msg.address & 0x7F,    # ADR (low)
376            msg.speed & 0x7F,
377            msg.dirf & 0x7F,
378            msg.trk & 0x7F,
379            msg.ss2 & 0x7F,      # SS2 — hours for slot 123, STAT2 elsewhere
380            (msg.address >> 7) & 0x7F,  # ADR2 (high)
381            msg.snd & 0x7F,
382            0x00,                # ID1
383            0x00,                # ID2
384        ])
385        return append_checksum(body)
386
387    if isinstance(msg, OpcUnknown):
388        body = bytes([msg.opcode]) + msg.payload
389        return append_checksum(body)
390
391    raise NotImplementedError(f"encode not implemented for opcode 0x{op:02X}")
