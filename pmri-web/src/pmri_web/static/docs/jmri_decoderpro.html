 86class DecoderCatalog:
 87    """Fast metadata index + lazy full-definition loader.
 88
 89    Two internal maps:
 90
 91    * ``_metadata`` — ``DecoderKey → DecoderIndexEntry`` from the fast scan
 92    * ``_cache``    — ``DecoderKey → DecoderDefinition`` filled on demand
 93
 94    ``programmers`` holds the standard programmer definitions
 95    (``Comprehensive.xml``, ``Basic.xml``, ...) parsed from the sibling
 96    ``programmers/`` dir. Those are small and load eagerly.
 97    """
 98
 99    def __init__(self) -> None:
100        self._metadata: dict[DecoderKey, DecoderIndexEntry] = {}
101        self._cache: dict[DecoderKey, DecoderDefinition] = {}
102        self.programmers: dict[str, ProgrammerConfig] = {}
103        # NMRA CV8 manufacturer ID → mfg name, loaded from
104        # ``decoderIndex.xml``'s ``<mfgList>`` at :meth:`from_directory`.
105        # Falls back to :data:`_NMRA_MFG_IDS` when the catalog wasn't
106        # loaded from a JMRI xml tree (tests, standalone probes).
107        self.mfg_ids: dict[int, str] = {}
108
109    # ----- construction ---------------------------------------------------

111    @classmethod
112    def from_directory(cls, root: str | pathlib.Path) -> DecoderCatalog:
113        """Build the catalog by scanning *root* for decoder XML files.
114
115        Only the identity fields are extracted at this stage — variables
116        + panes are parsed lazily via :meth:`find`. The sibling
117        ``programmers/`` directory (if present) is loaded eagerly since
118        it's small.
119        """
120        catalog = cls()
121        for entry in iter_decoder_index(root):
122            catalog._metadata[DecoderKey(entry.family, entry.model)] = entry
123        # decoderIndex.xml sits at the ``xml/`` root next to the
124        # ``decoders/`` dir and carries the authoritative NMRA
125        # manufacturer table (currently 179 entries). Load it if
126        # present so identify can map any CV8 the decoder returns.
127        index_path = pathlib.Path(root).parent / "decoderIndex.xml"
128        if index_path.is_file():
129            catalog.mfg_ids = _load_mfg_ids(index_path)
130        programmers_dir = pathlib.Path(root).parent / "programmers"
131        if programmers_dir.is_dir():
132            catalog.programmers = {
133                pc.name: pc for pc in iter_programmer_configs(programmers_dir)
134            }
135            _log.info(
136                "DecoderCatalog: indexed %d entries + %d programmer configs",
137                len(catalog), len(catalog.programmers),
138            )
139        else:
140            _log.info("DecoderCatalog: indexed %d entries", len(catalog))
141        return catalog

143    @classmethod
144    def from_files(cls, files: list[DecoderFile]) -> DecoderCatalog:
145        """Build the catalog from already-parsed :class:`DecoderFile`
146        objects. Used by tests + callers that want eager loading.
147
148        Populates both the metadata index and the definition cache so
149        subsequent :meth:`find` calls don't trigger any file I/O.

151        catalog = cls()
152        for f in files:

156    def add_index_entry(self, entry: DecoderIndexEntry) -> None:
157        """Register a scanned :class:`DecoderIndexEntry` in the metadata
158        index. Full-definition parse is deferred to first :meth:`find`.
159
160        Used by the web catalog-load worker so it can drive the
161        thermometer at scan time rather than parse time.
162        """
163        self._metadata[DecoderKey(entry.family, entry.model)] = entry

165    def _absorb(self, df: DecoderFile) -> None:
166        """Populate both metadata and cache from an already-parsed file."""
167        if df.family is None:
168            return
169        for model in df.family.models:
170            key = DecoderKey(df.family.name, model.model)
171            self._metadata[key] = DecoderIndexEntry(
172                family=df.family.name,
173                mfg=df.family.mfg,
174                model=model.model,
175                source_file=df.path or pathlib.Path(""),
176                # Propagate the ``<model productID="…"/>`` attribute
177                # so identify_by_cv_extended can narrow past the
178                # family level. Missing on tests / older files stays
179                # empty string, which the extended-identify path
180                # treats as "no productID match, keep in fallback".
181                product_id=getattr(model, "product_id", ""),
182            )
183            self._cache[key] = DecoderDefinition(
184                family=df.family,
185                model=model,
186                variables=tuple(df.variables),
187                panes=tuple(df.panes),
188                source_file=df.path,
189            )
190

193    def __len__(self) -> int:
194        return len(self._metadata)

197        """Iterate metadata entries — fast, no full parse.

199        Callers that need variables/panes should use :meth:`find` on the

210    def _lazy_load(self, key: DecoderKey) -> DecoderDefinition | None:
211        """Return the cached definition, parsing the source file on first hit."""
212        if key not in self._metadata:
213            return None
214        cached = self._cache.get(key)
215        if cached is not None:
216            return cached
217        meta = self._metadata[key]
218        source = meta.source_file
219        try:
220            df = load_decoder(source)
221        except Exception as e:  # noqa: BLE001
222            _log.warning(
223                "DecoderCatalog: lazy-load %s (%s/%s) failed: %s",
224                source, key.family, key.model, e,
225            )
226            return None
227        # One decoder file typically declares many models — cache them
228        # all, so a load triggered by find(F, M1) also serves find(F, M2)
229        # without re-parsing the file.
230        self._absorb(df)
231        return self._cache.get(key)

251    def identify_by_cv(
252        self, cv7_version: int, cv8_mfg: int,
253    ) -> list[DecoderIndexEntry]:
254        """Return metadata entries consistent with CV7/CV8 values.
255
256        Metadata-only (no lazy load): iterating a full catalog and
257        parsing every candidate file is prohibitively slow on modest
258        hardware — the JMRI decoder tree has ~2000 files, each with
259        xi:include fragments to resolve. Instead we filter by mfg alone
260        (from :attr:`mfg_ids` when the catalog was loaded from a JMRI
261        xml tree, or the small stopgap :data:`_NMRA_MFG_IDS` when it
262        wasn't) and let the caller pick a specific entry to fully load
263        via :meth:`find`.
264
265        Raises :class:`UnknownDecoderIdentityError` when *cv8_mfg* isn't
266        a known NMRA manufacturer ID — that includes the NCE NoAck
267        sentinel byte (``0x30`` = 48) that a failed programming-track
268        read returns. Surfacing the error is much friendlier than
269        silently returning "no matches" or, worse, iterating the whole
270        catalog when the read failed.
271        """
272        mfg_name = self.mfg_ids.get(cv8_mfg) or _NMRA_MFG_IDS.get(cv8_mfg)
273        if mfg_name is None:
274            raise UnknownDecoderIdentityError(cv7_version, cv8_mfg)
275        # Match case-insensitively AND on the leading token so
276        # "SoundTraxx (Throttle-Up)" matches entries stored as either
277        # form — decoderIndex.xml uses the parenthesised name; some
278        # older decoder files use the short form.
279        needle = mfg_name.split(" (")[0].lower()
280        # Filter by CV7 within the model's declared version range.
281        # JMRI decoder XMLs use ``lowVersionID`` / ``highVersionID`` on
282        # ``<family>`` (and less often on ``<model>``) to gate which
283        # decoder revisions a definition applies to. Models with no
284        # declared range stay wildcard (0..255) so old files without
285        # the attributes aren't hidden from the identify list.
286        return [
287            entry for entry in self._metadata.values()
288            if needle in entry.mfg.lower()
289            and entry.low_version_id <= cv7_version <= entry.high_version_id
290        ]
291
292    def identify_by_cv_extended(
293        self, cv7_version: int, cv8_mfg: int, product_id: int | None,
294    ) -> list[DecoderIndexEntry]:
295        """Same as :meth:`identify_by_cv`, then narrow by *product_id*.
296
297        When *product_id* is ``None`` (no probe run, or the probe
298        determined the model doesn't populate the productID CVs),
299        this returns the same list as :meth:`identify_by_cv` —
300        the caller shows the family-level picker and the user
301        chooses the model.
302
303        When *product_id* is an int, entries whose
304        :attr:`~jmri_core.persistence.decoder_xml.DecoderIndexEntry.product_id`
305        (the ``<model productID="…"/>`` XML attribute) matches are
306        preferred. JMRI stores comma-separated productID lists on
307        one model when the same XML applies to several variants —
308        we split and match against each. Entries without a
309        productID attribute are always kept in the fallback list
310        so no CV7/CV8 match is ever lost.
311
312        If any entry has a productID match, only those matches are
313        returned. If none do, we fall back to the CV7/CV8-only list
314        — this handles cases like a catalog scan that missed newer
315        productIDs the operator's decoder actually reports.
316        """
317        base = self.identify_by_cv(cv7_version, cv8_mfg)
318        if product_id is None:
319            return base
320        product_id_str = str(product_id)
321        matches = [
322            entry for entry in base
323            if _model_matches_product_id(entry.product_id, product_id_str)
324        ]
325        return matches if matches else base

Metadata-only (no lazy load): iterating a full catalog and parsing every candidate file is prohibitively slow on modest hardware — the JMRI decoder tree has ~2000 files, each with xi:include fragments to resolve. Instead we filter by mfg alone (from mfg_ids when the catalog was loaded from a JMRI xml tree, or the small stopgap _NMRA_MFG_IDS when it wasn't) and let the caller pick a specific entry to fully load via find().

Surfacing the error is much friendlier than silently returning "no matches" or, worse, iterating the whole catalog when the read failed.

113    """Build the ordered tab list for a loco's programmer view.
114

116
117    1. Synthetic ``Roster Entry`` tab always first.
118    2. Standard tabs from *programmer*, filtered against
119       ``_STANDARD_TAB_ORDER``. Any standard tab the programmer doesn't
120       declare is skipped.
121    3. Any decoder-declared pane from *decoder_panes* that isn't already
122       covered by name lands after the standard set.
123    4. Empty panes are removed unless *show_empty* forces them in
124       (mirrors JMRI's ``showEmptyPanes`` semantics; default follows the
125       programmer's own setting when *show_empty* is None).
126    """
127    if show_empty is None and programmer is not None:
128        show_empty = programmer.show_empty_panes.lower() == "yes"
129    if show_empty is None:

131

133
134    out: list[ComposedPane] = [
135        ComposedPane(
136            name=ROSTER_ENTRY_NAME,
137            slug=ROSTER_ENTRY_SLUG,
138            source="roster",
139            is_synthetic=True,
140        )
141    ]
142
143    # Decoder-declared panes take priority over the standard programmer
144    # pane with the same name — SoundTraxx Tsunami2 ships its own
145    # "Function Map" pane (Extended Function Map) that replaces
146    # Comprehensive.xml's generic ``<fnmapping/>`` matrix, ESU/Zimo do
147    # the same for their model-specific tabs, etc. Index the
148    # decoder-declared panes by name up front so the standard-pane pass
149    # can substitute them in place, preserving the standard tab order.
150    decoder_by_name: dict[str, DecoderPane] = {}
151    include_decoder_panes = programmer is None or programmer.decoder_file_panes

162                programmer_by_name.setdefault(pane.name, pane)
163

165    for standard_name in _STANDARD_TAB_ORDER:

167        if pane is None:

170        override = decoder_by_name.pop(standard_name, None)
171        if override is not None:
172            composed = _compose_one(override, "decoder", decoder_items)
173        else:
174            composed = _compose_one(pane, "programmer", decoder_items)
175        if composed.is_empty and not show_empty:
176            continue

178        out.append(composed)
179

181    # append after the standard block so downstream users still see them.
182    for pane in programmer_by_name.values():

184        if override is not None:

async def probe_tcs( session: ProgrammingSession, mfg_id: int, model_id: int) -> int | None: 
151async def probe_tcs(
152    session: ProgrammingSession, mfg_id: int, model_id: int,
153) -> int | None:
154    """Train Control Systems (TCS) productID probe.
155
156    TCS uses CV249 as the primary ID; older decoders (< 129) stop
157    there. Newer decoders extend the ID across CV248 / CV111 / CV110
158    with a family-specific packing formula (from
159    ``IdentifyDecoder.java`` lines 227..426).
160
161    Short form (CV249 < 129)::
162
163        productID = CV249
164
165    Extended form (CV249 ≥ 129)::
166
167        lowest = CV249; low = CV248; high = CV111; highest = CV110
168        # Which packing applies depends on ``model_id`` (from CV7) and
169        # the observed lowest byte — matches the branching in
170        # ``IdentifyDecoder.test7`` verbatim.
171    """
172    lowest = await session.programmer.read_cv(249)
173    if lowest < 129:
174        return lowest
175
176    low = await session.programmer.read_cv(248)
177    high = await session.programmer.read_cv(111)
178    highest = await session.programmer.read_cv(110)
179
180    # See ``IdentifyDecoder.test7`` (Java lines 414..425) for the
181    # authoritative branching. Kept as a nested if/else to make the
182    # correspondence with the Java source obvious to a reader who
183    # cross-checks; a table-driven form would obscure the mapping.
184    if (
185        (129 <= lowest <= 135 and low == 5)
186        or model_id >= 5
187    ):
188        if lowest == 180 and model_id == 5:
189            return lowest + low * 256
190        return (
191            lowest + low * 256 + high * 65536 + highest * 16777216
192        )
193    if (
194        (129 <= lowest <= 135)
195        or (170 <= lowest <= 172)
196        or lowest == 180
197    ) and model_id == 4:
198        return lowest + low * 256
199    return lowest
