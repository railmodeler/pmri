jmri_nce.messages

NCE commands and replies.

Each command is a fixed-shape request → fixed-shape reply pair. We model this declaratively with NceCommand and a small set of factory functions that build commands for the opcodes the rest of the package needs. New commands can be added without touching the traffic controller.

Every factory in this file must trace to a specific JMRI Java NceMessage or NceBinaryCommand factory. The wire format is audit-checked against JMRI's NceMessage.java opcode constants and NceMonBinaryBundle.properties. Do not add a factory without citing the JMRI source it mirrors.

 41@dataclass(frozen=True, slots=True)
 42class NceCommand:
 43    """One outbound request.
 44
 45    NCE has two distinct reply shapes:
 46
 47    * **Action commands** (power on/off, accessory throw, write CV, ...)
 48      reply with a single ACK byte (``!`` on success). ``reply_len`` is
 49      the number of *data* bytes that precede the ACK — usually 0.
 50    * **Read commands** (read memory, read AIU, ...) reply with exactly
 51      *reply_len* bytes of raw data, no trailing ACK. Set ``is_read=True``
 52      and the traffic controller will not consume an ACK byte after.
 53    """
 54
 55    name: str
 56    request: bytes
 57    reply_len: int = 0
 58    timeout_s: float = 0.5
 59    is_read: bool = False
 60
 61
 62@dataclass(frozen=True, slots=True)
 63class NceReply:
 64    """A successful response. ``data`` excludes the ACK byte itself."""
 65
 66    command: NceCommand
 67    data: bytes
 68
 69
 70class NceCommandError(Exception):
 71    """The command station replied with a non-OK ACK byte."""
 72
 73    def __init__(self, command: NceCommand, ack: bytes, data: bytes) -> None:
 74        super().__init__(
 75            f"NCE command {command.name!r} failed: ack={ack!r} data={data!r}"
 76        )
 77        self.command = command
 78        self.ack = ack
 79        self.data = data

122def write_memory(address: int, data: bytes) -> NceCommand:
123    """Write *data* (1..16 bytes) starting at the 16-bit *address*.
124
125    JMRI: ``NceBinaryCommand.accMemoryWriteN`` (opcode 0x8E, WRITE_N_CMD).
126
127    IMPORTANT: The CS expects a **fixed 20-byte request** regardless of
128    payload length — JMRI Java allocates ``new byte[20]`` and fills only
129    the first ``4 + data.length`` bytes, leaving the rest zero-padded.
130    We match that shape byte-for-byte; without the pad the CS waits
131    forever for the remaining bytes and the ACK never arrives.
132    """
133    if not 1 <= len(data) <= 16:
134        raise ValueError(f"write length must be 1..16, got {len(data)}")
135    header = bytes([
136        OP_WRITE_N_MEM,
137        (address >> 8) & 0xFF, address & 0xFF,
138        len(data) & 0xFF,
139    ])
140    padding = bytes(20 - len(header) - len(data))
141    return NceCommand(
142        name=f"WRITE_N_MEM({address:#06x},{len(data)})",
143        request=header + data + padding,
144    )

172def nce_loco_cmd(
173    loco_address: int, *, long_address: bool, sub_cmd: int, data: int
174) -> NceCommand:
175    """Throttle command via NCE binary ``nceLocoCmd``.
176
177    JMRI: ``NceBinaryCommand.nceLocoCmd`` (opcode 0xA2, LOCO_CMD).
178
179    Modern firmware (commandOptions >= 10) uses this 5-byte command for
180    speed, direction, and function changes.
181
182        [0xA2] [addr_hi] [addr_lo] [sub_cmd] [data]   (5 bytes)
183
184    For long addresses, the wire address is ``loco_address + 0xC000``
185    so the CS can disambiguate from short addresses with the same value.
186
187    *sub_cmd* values come from :mod:`jmri_nce.opcodes`:
188
189    * ``LOCO_SUB_128_FWD`` / ``LOCO_SUB_128_REV`` — 128-step speed
190    * ``LOCO_SUB_ESTOP_FWD`` / ``LOCO_SUB_ESTOP_REV`` — emergency stop
191    * ``LOCO_SUB_FUNC_GROUP_1..5`` — function bitmasks
192    """
193    if not 0 <= sub_cmd <= 23:
194        raise ValueError(f"sub_cmd must be 0..23, got {sub_cmd}")
195    if not 0 <= data <= 255:
196        raise ValueError(f"data byte must be 0..255, got {data}")
197    wire = loco_address + LONG_ADDRESS_OFFSET if long_address else loco_address
198    return NceCommand(
199        name=f"LOCO({loco_address}{'L' if long_address else 'S'},sub={sub_cmd},data={data})",
200        request=bytes([
201            OP_NCE_LOCO_CMD,
202            (wire >> 8) & 0xFF,
203            wire & 0xFF,
204            sub_cmd & 0xFF,
205            data & 0xFF,
206        ]),
207    )

238        request=bytes([OP_ENTER_PROG_MODE]),

293def ops_mode_write_cv(
294    loco_address: int, cv: int, value: int, *, long_address: bool = False,
295) -> NceCommand:
296    """Ops-mode CV byte-write (Programming On the Main) to a specific loco.
297
298    JMRI: ``NceMessage.sendPacketMessage`` builds the equivalent wrapped
299    packet in ``NceProgrammer.writeCV`` when ops mode is selected.
300
301    Wire form: build a raw NMRA CV Access instruction packet, then wrap
302    it in the appropriate ``QUEUEn_TMP_CMD`` envelope where ``n`` is the
303    packet's byte length (opcode ``0x93 + (packet_len - 3)`` covers
304    3..6-byte packets, per :data:`OP_QUEUE_PACKET_BASE`). NCE has no
305    native ops-mode CV *read* (RailCom is out of PowerPro's scope) — do
306    not add ``ops_mode_read_cv``; use direct mode for verified reads.
307
308    NMRA S-9.2.1 CV Access — Long Form / Byte Write:
309
310        <address bytes>  1110 CCVV   VVVVVVVV   DDDDDDDD  <XOR>
311
312    * short address (1..127): 1 address byte ``0AAAAAAA``
313    * long address (128..9999): 2 address bytes ``11AAAAAA AAAAAAAA``
314    * ``CC = 11`` → byte-write (10 = verify, 01 = bit-manipulate)
315    * ``VVVVVVVVVV`` = CV number minus 1, 10-bit
316    * ``DDDDDDDD`` = value
317    * trailing XOR of all preceding bytes
318
319    Reply: single ACK byte from the CS (``!`` on success). The packet
320    goes on the wire from the CS; no decoder ack round-trip.
321    """
322    if not 1 <= cv <= 1024:
323        raise ValueError(f"CV must be 1..1024, got {cv}")
324    if not 0 <= value <= 255:
325        raise ValueError(f"value must be 0..255, got {value}")
326
327    if long_address:
328        if not 128 <= loco_address <= 9999:
329            raise ValueError(
330                f"long address must be 128..9999, got {loco_address}"
331            )
332        # 14-bit long address in the DCC packet: top 6 bits go into the
333        # low 6 bits of 0xC0, low 8 bits stand alone.
334        addr_bytes = [
335            0xC0 | ((loco_address >> 8) & 0x3F),
336            loco_address & 0xFF,
337        ]
338    else:
339        if not 1 <= loco_address <= 127:
340            raise ValueError(
341                f"short address must be 1..127, got {loco_address}"
342            )
343        addr_bytes = [loco_address & 0x7F]
344
345    # NMRA encodes CV number minus one, spread over two bytes: 2 high
346    # bits ride the instruction opcode, 8 low bits follow.
347    cv_index = cv - 1
348    inst_hi = 0xEC | ((cv_index >> 8) & 0x03)   # 1110_1100 with CC=11 + VV
349    inst_lo = cv_index & 0xFF
350
351    packet = bytes(addr_bytes + [inst_hi, inst_lo, value & 0xFF])
352    xor = 0
353    for b in packet:
354        xor ^= b
355    dcc_packet = packet + bytes([xor])
356
357    # QUEUEn_TMP_CMD wire form (verified against JMRI Java NCE Command
358    # Monitor trace on real PowerPro hardware, 2026-07-25):
359    #
360    #   [queue_opcode] [send_count] [dcc_packet_bytes...]
361    #
362    # * ``queue_opcode`` = 0x93..0x96 (packet length 3..6)
363    # * ``send_count``   = number of times the CS retransmits the
364    #                      packet on the wire. NMRA POM recommends
365    #                      3–5 repeats for reliability; JMRI Java uses
366    #                      5, which we mirror. The CS reads this byte
367    #                      as part of the command — earlier versions
368    #                      of this code omitted it, so the CS would
369    #                      read our first packet byte as the send
370    #                      count and then time out ~5 s waiting for a
371    #                      packet byte that never came, returning
372    #                      NoAck (``0``).
373    # * ``dcc_packet``   = 3..6 bytes: address + CV Access instruction
374    #                      + XOR checksum, per NMRA S-9.2.1.
375    packet_len = len(dcc_packet)
376    if not 3 <= packet_len <= 6:
377        # Defensive: the packet builder should never produce a length
378        # outside 3..6, but if a future edit changes the encoding this
379        # catches the mistake before the wrong opcode goes out.
380        raise ValueError(
381            f"internal error: DCC packet length {packet_len} outside 3..6",
382        )
383    queue_opcode = OP_QUEUE_PACKET_BASE + (packet_len - 3)
384    send_count = 5   # matches JMRI Java default
385
386    return NceCommand(
387        name=f"WR_CV_OPS({loco_address}{'L' if long_address else 'S'},{cv},{value})",
388        request=bytes([queue_opcode, send_count]) + dcc_packet,
389        # Ops-mode writes are fire-and-forget on the wire; only the CS
390        # queue-accept ACK ('!') comes back. No data bytes.
391        #
392        # ``timeout_s`` is 5.0 s — matches JMRI Java's timeout for
393        # equivalent operations. NCE PowerPro's actual queue-then-ack
394        # latency depends on the CS's current DCC output load: with
395        # active throttles + fast-clock ticks + accessory refresh, we
396        # observed >2 s on real hardware even after bumping from the
397        # 0.5 s default. 5 s gives comfortable headroom; a real hang
398        # (packet dropped, CS off-line) still surfaces before the user
399        # gives up on the batch.
400        timeout_s=5.0,
401    )

404def read_clock() -> NceCommand:
405    """Read the CS fast clock (16 bytes at ``CS_CLOCK_ADDR``).
406
407    JMRI: ``NceClockControl.issueReadOnlyRequest`` uses
408    ``NceBinaryCommand.accMemoryRead(getClockAddr())`` (opcode 0x8F).
409    Wire request: 3 bytes ``[0x8F, addr_hi, addr_lo]``; reply is 16
410    data bytes. Passive read — safe to send even when the clock isn't
411    running.
412
413    Reply byte layout mirrors ``NceClockControl.readClockPacket``:
414
415    * ``[0]`` scale — CS-internal divisor, ``ratio = 250 / scale``
416    * ``[1]`` tick
417    * ``[2]`` seconds (binary, not BCD)
418    * ``[3]`` minutes
419    * ``[4]`` hours (1..12 in 12h mode; 0..23 in 24h)
420    * ``[5]`` AM/PM flag (ASCII ``'A'`` = 65 → AM)
421    * ``[6]`` 12/24 mode (1 = 24h, else 12h)
422    * ``[13]`` status
423
424    1.5 s timeout (vs the 0.5 s default) leaves headroom on slow USB-serial
425    adapters where a tightly-scheduled read can otherwise time out while
426    its 16 bytes are still in flight.
427    """
428    return NceCommand(
429        name="READ_CLOCK",
430        request=bytes([
431            OP_READ_16_MEM,
432            (CS_CLOCK_ADDR >> 8) & 0xFF, CS_CLOCK_ADDR & 0xFF,
433        ]),
434        reply_len=16,
435        is_read=True,
436        timeout_s=1.5,
437    )

440def write_clock_time(hours: int, minutes: int, seconds: int) -> NceCommand:
441    """Write hh/mm/ss to the CS clock.
442
443    JMRI: ``NceClockControl.issueClockSetMem`` writes ``[ss, mm, hh]`` at
444    ``CS_CLOCK_ADDR + 2`` via ``accMemoryWriteN`` (opcode 0x8E, WRITE_N_CMD).
445    Delegates to :func:`write_memory` so the fixed-20-byte wire pad is
446    applied in one place. 1.5 s timeout mirrors the clock read.
447    """
448    if not 0 <= hours <= 23:
449        raise ValueError(f"hours must be 0..23, got {hours}")
450    if not 0 <= minutes <= 59:
451        raise ValueError(f"minutes must be 0..59, got {minutes}")
452    if not 0 <= seconds <= 59:
453        raise ValueError(f"seconds must be 0..59, got {seconds}")
454    payload = bytes([seconds & 0xFF, minutes & 0xFF, hours & 0xFF])
455    inner = write_memory(CS_CLOCK_ADDR + 2, payload)
456    # Reuse the well-formed request but with a clock-specific name + timeout.
457    return NceCommand(
458        name=f"WRITE_CLOCK_TIME({hours:02d}:{minutes:02d}:{seconds:02d})",
459        request=inner.request,
460        timeout_s=1.5,
461    )

def ops_mode_write_cv( loco_address: int, cv: int, value: int, *, long_address: bool = False) -> NceCommand: 
294def ops_mode_write_cv(
295    loco_address: int, cv: int, value: int, *, long_address: bool = False,
296) -> NceCommand:
297    """Ops-mode CV byte-write (Programming On the Main) to a specific loco.
298
299    JMRI: ``NceMessage.sendPacketMessage`` builds the equivalent wrapped
300    packet in ``NceProgrammer.writeCV`` when ops mode is selected.
301
302    Wire form: build a raw NMRA CV Access instruction packet, then wrap
303    it in the appropriate ``QUEUEn_TMP_CMD`` envelope where ``n`` is the
304    packet's byte length (opcode ``0x93 + (packet_len - 3)`` covers
305    3..6-byte packets, per :data:`OP_QUEUE_PACKET_BASE`). NCE has no
306    native ops-mode CV *read* (RailCom is out of PowerPro's scope) — do
307    not add ``ops_mode_read_cv``; use direct mode for verified reads.
308
309    NMRA S-9.2.1 CV Access — Long Form / Byte Write:
310
311        <address bytes>  1110 CCVV   VVVVVVVV   DDDDDDDD  <XOR>
312
313    * short address (1..127): 1 address byte ``0AAAAAAA``
314    * long address (128..9999): 2 address bytes ``11AAAAAA AAAAAAAA``
315    * ``CC = 11`` → byte-write (10 = verify, 01 = bit-manipulate)
316    * ``VVVVVVVVVV`` = CV number minus 1, 10-bit
317    * ``DDDDDDDD`` = value
318    * trailing XOR of all preceding bytes
319
320    Reply: single ACK byte from the CS (``!`` on success). The packet
321    goes on the wire from the CS; no decoder ack round-trip.
322    """
323    if not 1 <= cv <= 1024:
324        raise ValueError(f"CV must be 1..1024, got {cv}")
325    if not 0 <= value <= 255:
326        raise ValueError(f"value must be 0..255, got {value}")
327
328    if long_address:
329        if not 128 <= loco_address <= 9999:
330            raise ValueError(
331                f"long address must be 128..9999, got {loco_address}"
332            )
333        # 14-bit long address in the DCC packet: top 6 bits go into the
334        # low 6 bits of 0xC0, low 8 bits stand alone.
335        addr_bytes = [
336            0xC0 | ((loco_address >> 8) & 0x3F),
337            loco_address & 0xFF,
338        ]
339    else:
340        if not 1 <= loco_address <= 127:
341            raise ValueError(
342                f"short address must be 1..127, got {loco_address}"
343            )
344        addr_bytes = [loco_address & 0x7F]
345
346    # NMRA encodes CV number minus one, spread over two bytes: 2 high
347    # bits ride the instruction opcode, 8 low bits follow.
348    cv_index = cv - 1
349    inst_hi = 0xEC | ((cv_index >> 8) & 0x03)   # 1110_1100 with CC=11 + VV
350    inst_lo = cv_index & 0xFF
351
352    packet = bytes(addr_bytes + [inst_hi, inst_lo, value & 0xFF])
353    xor = 0
354    for b in packet:
355        xor ^= b
356    dcc_packet = packet + bytes([xor])
357
358    # QUEUEn_TMP_CMD wire form (verified against JMRI Java NCE Command
359    # Monitor trace on real PowerPro hardware, 2026-07-25):
360    #
361    #   [queue_opcode] [send_count] [dcc_packet_bytes...]
362    #
363    # * ``queue_opcode`` = 0x93..0x96 (packet length 3..6)
364    # * ``send_count``   = number of times the CS retransmits the
365    #                      packet on the wire. NMRA POM recommends
366    #                      3–5 repeats for reliability; JMRI Java uses
367    #                      5, which we mirror. The CS reads this byte
368    #                      as part of the command — earlier versions
369    #                      of this code omitted it, so the CS would
370    #                      read our first packet byte as the send
371    #                      count and then time out ~5 s waiting for a
372    #                      packet byte that never came, returning
373    #                      NoAck (``0``).
374    # * ``dcc_packet``   = 3..6 bytes: address + CV Access instruction
375    #                      + XOR checksum, per NMRA S-9.2.1.
376    packet_len = len(dcc_packet)
377    if not 3 <= packet_len <= 6:
378        # Defensive: the packet builder should never produce a length
379        # outside 3..6, but if a future edit changes the encoding this
380        # catches the mistake before the wrong opcode goes out.
381        raise ValueError(
382            f"internal error: DCC packet length {packet_len} outside 3..6",
383        )
384    queue_opcode = OP_QUEUE_PACKET_BASE + (packet_len - 3)
385    send_count = 5   # matches JMRI Java default
386
387    return NceCommand(
388        name=f"WR_CV_OPS({loco_address}{'L' if long_address else 'S'},{cv},{value})",
389        request=bytes([queue_opcode, send_count]) + dcc_packet,
390        # Ops-mode writes are fire-and-forget on the wire; only the CS
391        # queue-accept ACK ('!') comes back. No data bytes.
392        #
393        # ``timeout_s`` is 5.0 s — matches JMRI Java's timeout for
394        # equivalent operations. NCE PowerPro's actual queue-then-ack
395        # latency depends on the CS's current DCC output load: with
396        # active throttles + fast-clock ticks + accessory refresh, we
397        # observed >2 s on real hardware even after bumping from the
398        # 0.5 s default. 5 s gives comfortable headroom; a real hang
399        # (packet dropped, CS off-line) still surfaces before the user
400        # gives up on the batch.
401        timeout_s=5.0,
402    )
