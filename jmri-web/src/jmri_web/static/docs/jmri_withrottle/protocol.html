jmri_withrottle.protocol

WiThrottle line protocol — parsing + encoding.

WiThrottle is a line-based ASCII protocol on TCP. Newline-terminated messages flow in both directions, with mostly self-describing first characters:

  • VN — protocol version
  • HT — server (railroad) name
  • RL — roster list
  • PPA — track power: PPA1 on, PPA0 off
  • PFT — fast clock
  • M{T,S}… — throttle multiplex commands; T is the primary throttle, S is the secondary on the same client
  • * — heartbeat tick from client
  • Q — quit

We implement enough of the spec to drive a loco from Engine Driver: acquire/release a throttle, set speed/direction/functions, e-stop, heartbeat. The fancier bits (consist building, signal cues, turnout/route control over WiThrottle, panel data, fast-clock sync) land in follow-up sprints.

References:

  • https://www.jmri.org/help/en/package/jmri/jmrit/withrottle/Protocol.shtml
  • Engine Driver's source (Android) for real-world client behaviour.

144def decode_line(line: str) -> object:
145    """Parse one WiThrottle inbound line. Returns one of the typed messages."""
146    line = line.rstrip("\r\n")
147    if not line:
148        return Unknown(raw="")
149    head = line[0]
150    if head == "*":
151        return Heartbeat()
152    if head == "Q":
153        return Quit()
154    if head == "N" and len(line) > 1:
155        return HelloName(name=line[1:])
156    if head == "H" and line.startswith("HU"):
157        return HelloHardwareId(hardware_id=line[2:])
158    if head == "M":
159        return _decode_throttle(line)
160    if line.startswith("PTA") and len(line) >= 4:
161        return TurnoutAction(action=line[3], system_name=line[4:])
162    return Unknown(raw=line)
163
164
165def _decode_throttle(line: str) -> object:
166    """``M{T,S}<op><addr><;><payload>``."""
167    if len(line) < 4:
168        return Unknown(raw=line)
169    throttle = line[1]
170    op = line[2]
171    rest = line[3:]
172    # The wire uses "<;>" (the literal three characters) as a separator.
173    sep = "<;>"
174    if sep in rest:
175        addr_token, payload = rest.split(sep, 1)
176    else:
177        addr_token, payload = rest, ""
178    if op == "+":
179        return AcquireLoco(throttle=throttle, address_token=addr_token, label=payload)
180    if op == "-":
181        return ReleaseLoco(throttle=throttle, address_token=addr_token)
182    if op == "A":
183        if not payload:
184            return Unknown(raw=line)
185        action = payload[0]
186        value = payload[1:]
187        return ThrottleAction(
188            throttle=throttle, address_token=addr_token,
189            action=action, value=value,
190        )
191    return Unknown(raw=line)

197def parse_address_token(token: str) -> tuple[int, bool]:
198    """Decode WiThrottle's address-token format.
199
200    * ``L<n>`` — explicit long address ``n``
201    * ``S<n>`` — explicit short address ``n``
202    * bare digits — NMRA convention: addresses ``1..127`` are short,
203      ``≥128`` are long. Older clients (and some legacy code paths in
204      Engine Driver itself) drop the prefix; matching JMRI's behaviour,
205      we infer the type from the numeric range so high-address locos
206      get the correct DCC encoding.
207
208    Returns ``(address, is_long)``.
209    """
210    if not token:
211        raise ValueError("empty address token")
212    if token[0] in ("L", "l"):
213        return int(token[1:]), True
214    if token[0] in ("S", "s"):
215        return int(token[1:]), False
216    addr = int(token)
217    # NMRA: short DCC range is 1..127; anything higher must be long.
218    return addr, addr >= 128

221def format_address_token(address: int, long_address: bool) -> str:

249def roster_list(entries: list[dict]) -> str:
250    """``RL<count>]\\[<id><}|><address>]\\[<long|short>]\\[<id>...``.
251
252    *entries* are roster-service dicts (id, dccAddress, longAddress).
253    """
254    parts: list[str] = [f"RL{len(entries)}"]
255    for e in entries:
256        eid = str(e.get("id") or "")
257        addr = str(e.get("dccAddress") or "")
258        long = "L" if e.get("longAddress") else "S"
259        # Wire form per JMRI's WiThrottle docs.
260        parts.append(f"{eid}}}|{{{addr}}}|{{{long}")
261    # JMRI uses "]\\[" (the literal three characters) between entries.
262    return "]\\[".join(parts)

299def throttle_function_labels(
300    throttle: str, address_token: str, labels: list[str] | None = None,
301) -> str:
302    """``M<mux>L<addr><;>]\\[label0]\\[label1]...]\\[label28]``.
303
304    JMRI's ``MultiThrottleController.sendFunctionLabels`` — this is the
305    line Engine Driver and WiThrottle iOS need after acquire so they can
306    render the F0..F28 button bar.
307
308    The label values matter more than they look:
309
310    * **Empty** labels are interpreted by Engine Driver as "no function
311      configured" and the bar stays hidden.
312    * **Missing line** works only for manual (``L<addr>``) acquires; for
313      roster (``E<id>``) acquires the client expects labels and hides
314      the bar when none arrive.
315
316    So we always send the line with non-empty defaults (``F0``..``F28``).
317    Pass explicit *labels* when you have roster data — anything falsy in
318    the list (empty string, ``None``) falls back to the default for that
319    slot.
320    """
321    if labels is None:
322        wire_labels = list(_DEFAULT_FUNCTION_LABELS)
323    else:
324        # Merge caller-supplied labels over defaults; empty slots keep
325        # the default so the client always sees something.
326        wire_labels = list(_DEFAULT_FUNCTION_LABELS)
327        for i, lbl in enumerate(labels[: MAX_FUNCTION + 1]):
328            if lbl:
329                wire_labels[i] = lbl
330    return f"M{throttle}L{address_token}<;>" + "".join(
331        f"]\\[{lbl}" for lbl in wire_labels
332    )

335def fast_clock(local_unix_seconds: int, rate: float) -> str:
336    """``PFT<local-unix-seconds><;><rate>``.
337
338    JMRI's ``FastClockController.sendFastTimeAndRate`` sends
339    ``(UTC_millis + TZ_offset) / 1000`` — the fast-clock time as if
340    it were a unix timestamp *in local time* (i.e. UTC unix seconds
341    biased by the local timezone offset so ``client_hour == local_hour``
342    when the client renders it as UTC).
343
344    A rate of ``0.0`` is the "stopped" signal — JMRI sends a second
345    ``PFT<time><;>0.0`` after the normal broadcast when the clock is
346    held; Engine Driver / WiThrottle iOS use that to pause their own
347    display.
348    """
349    return f"PFT{int(local_unix_seconds)}<;>{rate}"

373def turnout_list(entries: list[dict]) -> str:
374    """``PTL]\\[<sysName>}|{<userName>}|{<state>]\\[...``
375
376    JMRI's ``TurnoutController.sendList``. Each entry is a dict with:
377
378    * ``system_name`` — required, e.g. ``NT12``
379    * ``user_name`` — optional friendly label; empty string if absent
380    * ``state`` — one of :data:`TURNOUT_UNKNOWN/CLOSED/THROWN/INCONSISTENT`
381    """
382    parts: list[str] = ["PTL"]
383    for e in entries:
384        sys_name = str(e.get("system_name") or "")
385        user_name = str(e.get("user_name") or "")
386        state = int(e.get("state") or TURNOUT_UNKNOWN)
387        parts.append(f"]\\[{sys_name}}}|{{{user_name}}}|{{{state}")
388    return "".join(parts)

401def fast_clock_local_unix_seconds(
402    hours: int, minutes: int, seconds: int = 0,
403    *, now: object = None,
404) -> int:
405    """Convert an ``HH:MM:SS`` fast-clock time to the local-unix value
406    JMRI's ``PFT`` payload expects.
407
408    We fold today's date onto the given HH:MM:SS to build a full
409    datetime, then bias UTC unix seconds by the current timezone offset
410    (matching JMRI's ``(fastClockTime + timeZoneOffset) / 1000L``).
411    """
412    import datetime  # noqa: PLC0415
413    import time as _time  # noqa: PLC0415
414    ref = now if isinstance(now, datetime.datetime) else datetime.datetime.now()
415    ref = ref.replace(hour=hours, minute=minutes, second=seconds, microsecond=0)
416    utc_unix = ref.timestamp()
417    # `time.timezone` is seconds west of UTC (positive); we want the
418    # opposite sign. `altzone` applies when DST is in effect.
419    if _time.localtime(utc_unix).tm_isdst > 0 and _time.daylight:
420        offset = -_time.altzone
421    else:
422        offset = -_time.timezone
423    return int(utc_unix + offset)

def parse_address_token(token: str) -> tuple[int, bool]: 
198def parse_address_token(token: str) -> tuple[int, bool]:
199    """Decode WiThrottle's address-token format.
200
201    * ``L<n>`` — explicit long address ``n``
202    * ``S<n>`` — explicit short address ``n``
203    * bare digits — NMRA convention: addresses ``1..127`` are short,
204      ``≥128`` are long. Older clients (and some legacy code paths in
205      Engine Driver itself) drop the prefix; matching JMRI's behaviour,
206      we infer the type from the numeric range so high-address locos
207      get the correct DCC encoding.
208
209    Returns ``(address, is_long)``.
210    """
211    if not token:
212        raise ValueError("empty address token")
213    if token[0] in ("L", "l"):
214        return int(token[1:]), True
215    if token[0] in ("S", "s"):
216        return int(token[1:]), False
217    addr = int(token)
218    # NMRA: short DCC range is 1..127; anything higher must be long.
219    return addr, addr >= 128

def fast_clock_local_unix_seconds(hours: int, minutes: int, seconds: int = 0, *, now: object = None) -> int: 
402def fast_clock_local_unix_seconds(
403    hours: int, minutes: int, seconds: int = 0,
404    *, now: object = None,
405) -> int:
406    """Convert an ``HH:MM:SS`` fast-clock time to the local-unix value
407    JMRI's ``PFT`` payload expects.
408
409    We fold today's date onto the given HH:MM:SS to build a full
410    datetime, then bias UTC unix seconds by the current timezone offset
411    (matching JMRI's ``(fastClockTime + timeZoneOffset) / 1000L``).
412    """
413    import datetime  # noqa: PLC0415
414    import time as _time  # noqa: PLC0415
415    ref = now if isinstance(now, datetime.datetime) else datetime.datetime.now()
416    ref = ref.replace(hour=hours, minute=minutes, second=seconds, microsecond=0)
417    utc_unix = ref.timestamp()
418    # `time.timezone` is seconds west of UTC (positive); we want the
419    # opposite sign. `altzone` applies when DST is in effect.
420    if _time.localtime(utc_unix).tm_isdst > 0 and _time.daylight:
421        offset = -_time.altzone
422    else:
423        offset = -_time.timezone
424    return int(utc_unix + offset)
