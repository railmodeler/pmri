 68    async def _cs_ping_loop(self) -> None:
 69        """Send ``<s>`` every 2s so the CS keeps replying with status
 70        traffic — that's what the reachability monitor watches for."""
 71        import contextlib  # noqa: PLC0415
 72        try:
 73            while True:
 74                await asyncio.sleep(2.0)
 75                if self._tc is None:
 76                    return
 77                try:
 78                    await self._tc.send(DccexMessage(command="s", args=()))
 79                except Exception:  # noqa: BLE001
 80                    # Send may raise on a closed port; the monitor
 81                    # loop notices via the silence timer.
 82                    pass
 83        except asyncio.CancelledError:
 84            with contextlib.suppress(Exception):

 87    async def _cs_reachability_loop(self) -> None:
 88        """3-second silence rule against ``last_inbound_at``."""
 89        import contextlib  # noqa: PLC0415
 90        try:
 91            await asyncio.sleep(3.0)   # startup grace
 92            while True:
 93                await asyncio.sleep(1.0)
 94                if self._tc is None:
 95                    return
 96                last = self._tc.last_inbound_at
 97                if last is None:

 99                    continue
100                age = time.monotonic() - last
101                await self._set_cs_reachable(age <= 3.0)

104                return

 76    async def read_cv_ops(
 77        self, address: int, cv: int, *, timeout: float = 6.0,
 78    ) -> int:

 80
 81        Wire: ``<r cab cv>`` request → ``<r cab cv value>`` reply. The
 82        CS's ``r`` handler explicitly requires ``DCCWaveform::isRailcom()``
 83        to return true; if RailCom is disabled in the firmware config
 84        the request is silently dropped, so a caller sees this as a
 85        timeout rather than a distinctive error. The timeout message
 86        surfaces that possibility explicitly.
 87
 88        Failure (``value == -1``) surfaces as :class:`RuntimeError`
 89        matching the same-shaped failure on the service-mode path so
 90        callers can use one exception filter for both.
 91        """

 93            return await self._send_and_wait_ops_read(
 94                address=address, cv=cv, timeout=timeout,

 97    async def _send_and_wait_r(
 98        self, request: DccexMessage, *, expected_cv: int, timeout: float
 99    ) -> int:
100        """Await a 2-arg service-mode reply matching ``expected_cv``.
101
102        Accepts either ``<r cv value>`` (write ack + some firmwares'
103        read replies) or ``<v cv value>`` (canonical read reply per
104        DCC-EX upstream). Callers pass the CV they're expecting; a
105        reply for a different CV is ignored so an unrelated CS
106        message doesn't complete our future prematurely.
107        """
108        future: asyncio.Future[int] = asyncio.get_event_loop().create_future()
109
110        async def listener(msg: DccexMessage) -> None:
111            if future.done():
112                return
113            # Accept both letters — see the module docstring.
114            if msg.command not in ("r", "v") or len(msg.args) != 2:
115                return
116            try:
117                cv = int(msg.args[0])
118                value = int(msg.args[1])
119            except ValueError:
120                return
121            if cv != expected_cv:
122                return
123            if value < 0:
124                future.set_exception(
125                    RuntimeError(f"DCC-EX programming op failed on CV {cv}")
126                )
127            else:
128                future.set_result(value)
129
130        disconnect = self._tc.inbound.connect(listener)
131        try:
132            await self._tc.send(request)
133            return await asyncio.wait_for(future, timeout=timeout)
134        finally:
135            disconnect()
136
137    async def _send_and_wait_ops_read(
138        self, *, address: int, cv: int, timeout: float,
139    ) -> int:
140        """Send ``<r cab cv>`` and await ``<r cab cv value>`` (3 args).
141
142        The 3-arg shape lets the listener distinguish ops-mode
143        replies from service-mode replies unambiguously — a 2-arg
144        ``<r ...>`` is a write ack, a 3-arg ``<r ...>`` is a RailCom
145        read result.
146        """
147        future: asyncio.Future[int] = asyncio.get_event_loop().create_future()
148
149        async def listener(msg: DccexMessage) -> None:
150            if future.done():
151                return
152            if msg.command != "r" or len(msg.args) != 3:
153                return
154            try:
155                cab = int(msg.args[0])
156                got_cv = int(msg.args[1])
157                value = int(msg.args[2])
158            except ValueError:
159                return
160            if cab != address or got_cv != cv:
161                return
162            if value < 0:
163                future.set_exception(RuntimeError(
164                    f"DCC-EX ops-mode read failed on cab={address} CV={cv} "
165                    "(no RailCom response)"
166                ))
167            else:
168                future.set_result(value)
169
170        disconnect = self._tc.inbound.connect(listener)
171        try:
172            await self._tc.send(DccexMessage(
173                command="r", args=(str(address), str(cv)),
174            ))
175            return await asyncio.wait_for(future, timeout=timeout)
176        except asyncio.TimeoutError as e:

180                "verify the firmware was built with RAILCOM enabled and "
181                "that the decoder supports RailCom cutout."

183        finally:
