jmri_core

jmri_core — Python rewrite of JMRI's core domain model.

This package is pure logic with no I/O. It contains:

  • jmri_core.bean — NamedBean protocol and AbstractNamedBean mixin.
  • jmri_core.events — Signal, the async-friendly event bus that replaces JMRI's PropertyChangeListener.
  • jmri_core.manager — Manager generic, the per-type registry underlying every JMRI *Manager.
  • jmri_core.instance — InstanceManager, the service-locator container.
  • jmri_core.names — system-name parsing ("LT12" → connection prefix "L", type letter "T", address "12").
  • jmri_core.state — shared state enums.
  • jmri_core.beans — concrete bean classes (Turnout, Sensor, …).
  • jmri_core.hal — SystemConnection ABC + internal stub.
  • jmri_core.persistence — XML load/save for roster and config files.
  • jmri_core.plugins — entry-point discovery for drivers and extensions.

Hardware drivers and UI live in separate packages that depend on this one; this package never depends on them.

25
26    Construct one and pass it explicitly (or via dependency injection) into
27    components that need to resolve services. This avoids the global-state
28    pitfalls of JMRI's Java ``InstanceManager`` while keeping the locator
29    pattern that scripts and tools expect.
30    """
31
32    def __init__(self) -> None:
33        self._services: dict[type, object] = {}
34        self._factories: dict[type, Callable[[], object]] = {}
35
36    def register(self, iface: type[T], impl: T) -> None:
37        """Bind a concrete instance to *iface*. Replaces any prior binding."""
38        self._services[iface] = impl
39        # Clearing a factory ensures explicit register() wins over factory.
40        self._factories.pop(iface, None)
41
42    def register_factory(self, iface: type[T], factory: Callable[[], T]) -> None:
43        """Bind a zero-arg factory invoked on first :meth:`get`. Idempotent."""
44        # If something is already bound, prefer the factory only if no live
45        # instance exists.
46        self._factories[iface] = factory
47
48    def get(self, iface: type[T]) -> T:
49        """Return the bound implementation of *iface*. Build via factory if needed."""
50        if iface in self._services:
51            return self._services[iface]  # type: ignore[return-value]
52        if iface in self._factories:
53            built = self._factories[iface]()
54            self._services[iface] = built
55            return built  # type: ignore[return-value]
56        raise LookupError(f"No service registered for {iface.__name__}")

59        """Like :meth:`get` but returns ``None`` if no binding exists."""
60        try:
61            return self.get(iface)
62        except LookupError:
63            return None
64
65    def contains(self, iface: type) -> bool:
66        return iface in self._services or iface in self._factories
67
68    def remove(self, iface: type) -> None:
69        """Drop the binding for *iface*. No-op if absent."""
70        self._services.pop(iface, None)
71        self._factories.pop(iface, None)

 25class Manager[T: NamedBean]:
 26    """Per-type bean registry. Subclass and override :meth:`_create`.
 27
 28    Attributes set on the class describe the bean type:
 29
 30    * ``type_letter`` — single uppercase letter the bean class owns
 31      (e.g. ``"T"`` for Turnout). Must be set by subclass.
 32    * ``system_prefix`` — connection prefix (e.g. ``"L"``). Set per-instance

 38    def __init__(self, system_prefix: str) -> None:
 39        if not self.type_letter:
 40            raise TypeError(
 41                f"{type(self).__name__}.type_letter must be set on the class"

 45        self.system_prefix = system_prefix
 46        self._by_system: dict[str, T] = {}

 81    async def provide(self, name: str) -> T:
 82        """Get-or-create. Accepts either system name or address-only form.
 83
 84        Address-only example: passing ``"12"`` to a LocoNet turnout manager
 85        with ``system_prefix='L'`` resolves to ``"LT12"``. Passing the full
 86        system name ``"LT12"`` is also accepted.
 87        """
 88        if name in self._by_user:
 89            return self._by_user[name]
 90        sysname = self._normalize_name(name)
 91        if sysname in self._by_system:
 92            return self._by_system[sysname]
 93        bean = self._create(sysname)
 94        await self.register(bean)
 95        return bean
 96
 97    async def register(self, bean: T) -> None:
 98        """Add an externally constructed *bean* to the registry."""
 99        if bean.system_name in self._by_system:
100            raise ValueError(f"bean with system_name {bean.system_name!r} already registered")
101        self._by_system[bean.system_name] = bean
102        if bean.user_name:
103            if bean.user_name in self._by_user:
104                raise ValueError(f"user_name {bean.user_name!r} already in use")
105            self._by_user[bean.user_name] = bean
106        await self.added.emit(bean)
107
108    async def deregister(self, bean: T) -> None:
109        """Remove *bean* from the registry. No-op if not present."""
110        if bean.system_name not in self._by_system:
111            return
112        del self._by_system[bean.system_name]
113        if bean.user_name and self._by_user.get(bean.user_name) is bean:
114            del self._by_user[bean.user_name]
115        await self.removed.emit(bean)
116
117    # ---------------------------------------------------------------- internal
118
119    def _normalize_name(self, name: str) -> str:
120        """Turn *name* into a full system name. Accepts either form."""
121        if name.startswith(self.full_prefix):
122            return name
123        # Treat as bare address. Validate it parses cleanly.
124        return SystemName.make(self.system_prefix, self.type_letter, name)
125
126    def _create(self, system_name: str) -> T:
127        """Subclasses must override to construct their concrete bean type."""
128        raise NotImplementedError(
129            f"{type(self).__name__}._create must be implemented by subclasses"
130        )
131
132    # Convenience for tests / persistence ------------------------------------
133
134    def _set_user_name(self, bean: T, new_name: str | None) -> None:
135        """Rename *bean* maintaining the user-name index. Internal."""
136        # Remove old binding if any
137        old = bean.user_name
138        if old and self._by_user.get(old) is bean:
139            del self._by_user[old]
140        bean.user_name = new_name
141        if new_name:
142            if new_name in self._by_user and self._by_user[new_name] is not bean:
143                raise ValueError(f"user_name {new_name!r} already in use")
144            self._by_user[new_name] = bean

146    def to_dict(self) -> dict[str, Any]:

1945def _no_init_or_replace_init(self, *args, **kwargs):
1946    cls = type(self)
1947
1948    if cls._is_protocol:
1949        raise TypeError('Protocols cannot be instantiated')
1950
1951    # Already using a custom `__init__`. No need to calculate correct
1952    # `__init__` to call. This can lead to RecursionError. See bpo-45121.
1953    if cls.__init__ is not _no_init_or_replace_init:
1954        return
1955
1956    # Initially, `__init__` of a protocol subclass is set to `_no_init_or_replace_init`.
1957    # The first instantiation of the subclass will call `_no_init_or_replace_init` which
1958    # searches for a proper new `__init__` in the MRO. The new `__init__`
1959    # replaces the subclass' old `__init__` (ie `_no_init_or_replace_init`). Subsequent
1960    # instantiation of the protocol subclass will thus use the new
1961    # `__init__` and no longer call `_no_init_or_replace_init`.
1962    for base in cls.__mro__:
1963        init = base.__dict__.get('__init__', _no_init_or_replace_init)
1964        if init is not _no_init_or_replace_init:
1965            cls.__init__ = init
1966            break
1967    else:
1968        # should not happen
1969        cls.__init__ = object.__init__
1970
1971    cls.__init__(self, *args, **kwargs)

31class Signal[T]:
32    """An emitter with zero or more listeners. See module docstring."""
33
34    __slots__ = ("_listeners",)
35
36    def __init__(self) -> None:
37        self._listeners: list[Listener[T]] = []
38
39    def connect(self, listener: Listener[T]) -> Callable[[], None]:
40        """Register *listener*. Returns a callable that disconnects it.
41
42        ``listener`` may be a sync or async callable taking one argument.
43        Returns a zero-arg disconnect function.
44        """
45        self._listeners.append(listener)
46
47        def disconnect() -> None:
48            # Idempotent: removing twice is a no-op.
49            with contextlib.suppress(ValueError):
50                self._listeners.remove(listener)
51
52        return disconnect
53
54    def disconnect_all(self) -> None:
55        """Drop every listener. Useful in teardown."""
56        self._listeners.clear()
57
58    @property
59    def listener_count(self) -> int:
60        return len(self._listeners)
61
62    async def emit(self, payload: T) -> None:
63        """Deliver *payload* to every connected listener.
64
65        Sync listeners run inline. Async listeners are awaited sequentially in
66        registration order — emission preserves ordering, which beans rely on
67        for state-change semantics (commanded-state notifications must arrive
68        before known-state notifications). If you need concurrency, use
69        :meth:`emit_concurrent`.
70        """
71        first_error: BaseException | None = None
72        # Snapshot the list so listeners that disconnect during dispatch don't
73        # mutate the iteration.
74        for listener in list(self._listeners):
75            try:
76                result = listener(payload)
77                if inspect.iscoroutine(result):
78                    await result
79            except BaseException as e:  # noqa: BLE001 — re-raised below
80                if first_error is None:
81                    first_error = e
82        if first_error is not None:
83            raise first_error
84
85    async def emit_concurrent(self, payload: T) -> None:
86        """Fan out to every async listener concurrently via ``asyncio.gather``.
87
88        Sync listeners still run inline before dispatching the async batch.
89        Use only when listeners are explicitly independent.
90        """
91        async_results: list[Awaitable[None]] = []
92        for listener in list(self._listeners):
93            result = listener(payload)
94            if inspect.iscoroutine(result):
95                async_results.append(result)

64
65        Sync listeners run inline. Async listeners are awaited sequentially in
66        registration order — emission preserves ordering, which beans rely on
67        for state-change semantics (commanded-state notifications must arrive
68        before known-state notifications). If you need concurrency, use
69        :meth:`emit_concurrent`.

79            except BaseException as e:  # noqa: BLE001 — re-raised below
