jmri_server.auth

Authentication + authorization for the JSON API and web UI.

Turn-on: set JMRI_AUTH_ENABLED=1 in the environment before starting the server. With auth off (the default), the server behaves as it did before — every route open, no login required. That keeps existing single-user-on-LAN Pi deployments working without change.

Users live in <settings-root>/users.yaml — one entry per user with a bcrypt-hashed password + a personal API token + one of three roles (viewer / operator / admin). See ~jmri_server.auth.models.User.

Authentication mechanisms:

  • Session cookie (browser). Set by /login on password verify; carries a signed user identity. Cleared by /logout.
  • Bearer token (JSON API + WiThrottle). Long-lived per-user token in users.yaml; the client presents it in the Authorization: Bearer <token> header.

Authorization is by role:

  • viewer — can get / list any topic. No mutations.
  • operator — everything viewer can, plus post / put / delete on beans, throttles, programmer, roster, panels, LogixNG, scripts.
  • admin — everything, plus configProfile / beanTransfer / (future) configUser.

Bootstrap: with auth enabled but users.yaml missing, the ~jmri_server.auth.manager.AuthManager mints an admin user with a random 32-byte password + token, writes the plaintext password to <settings-root>/admin_password.txt (mode 0600), and logs a one-time WARNING telling the operator to pick it up. On next startup the file is left alone — deleting it manually is fine.

For the JMRI Java-API-compatible token flow, see the docstrings on ~jmri_server.auth.manager.AuthManager and the middleware.

 87    def login_password(
 88        self, username: str, password: str,
 89    ) -> LoginResult | None:
 90        """Verify *password* for *username*. Returns the login
 91        result on success, ``None`` on any failure (unknown user,
 92        wrong password, malformed hash). Never leaks which of the
 93        two failed via return value — the log line records it for
 94        the operator's audit."""
 95        user = self._users.by_username(username)
 96        if user is None:
 97            _log.info("auth: login failed for unknown user %r", username)
 98            return None
 99        if not verify_password(password, user.password_hash):
100            _log.info("auth: bad password for user %r", username)
101            return None
102        return LoginResult(user=user)
103
104    def login_token(self, token: str) -> LoginResult | None:
105        """Match a Bearer token against every user's personal
106        token. Returns the login result on success, ``None`` on
107        no match. Empty token → ``None`` without querying the
108        store."""
109        user = self._users.by_token(token)
110        if user is None:
111            return None
112        return LoginResult(user=user)

116    def mint_session_cookie(self, user: User) -> str:
117        """Encode *user* as a signed cookie value. The value round-
118        trips through :meth:`verify_session_cookie`; anything else
119        (tampered value, expired timestamp) returns ``None`` there."""
120        return self._serializer.dumps({"username": user.username})
121
122    def verify_session_cookie(self, cookie_value: str) -> User | None:
123        """Decode + validate. Returns the current :class:`User` for
124        the encoded username, or ``None`` for tampered / expired /
125        deleted-user cases.
126
127        Deleted-user handling matters: a user removed from
128        ``users.yaml`` should be locked out immediately, not on
129        session expiry. Re-checking the store on every verify is
130        cheap (dict lookup) and closes that hole."""
131        if not cookie_value:
132            return None
133        try:
134            payload = self._serializer.loads(
135                cookie_value, max_age=SESSION_MAX_AGE_SECONDS,
136            )
137        except SignatureExpired:
138            return None
139        except BadSignature:
140            return None
141        if not isinstance(payload, dict):
142            return None
143        username = payload.get("username")
144        if not isinstance(username, str):
145            return None
146        # Re-fetch so a role change (or deletion) after cookie
147        # mint takes effect immediately.
148        return self._users.by_username(username)

167    def add_user(
168        self,
169        username: str,
170        password: str,
171        *,
172        role: Role = Role.VIEWER,
173        token: str | None = None,
174    ) -> User:
175        """Create a new user. Persists to disk atomically.
176
177        ``token`` defaults to a fresh random one — a client (mobile
178        throttle, etc.) can start using it immediately. Pass an
179        explicit empty string to create a login-only account.
180        """
181        if not username:
182            raise ValueError("username must be non-empty")
183        if self._users.by_username(username) is not None:
184            raise ValueError(f"user {username!r} already exists")
185        actual_token = (
186            token if token is not None else secrets.token_urlsafe(32)
187        )
188        user = User(
189            username=username,
190            password_hash=hash_password(password),
191            role=role,
192            token=actual_token,
193        )
194        self._users.users.append(user)
195        self._persist()
196        return user
197
198    def remove_user(self, username: str, *, current_username: str) -> None:
199        """Delete *username*. Two guards:
200
201        * Cannot delete yourself — refuses when
202          ``username == current_username`` so an admin can't
203          accidentally lock the whole install out.
204        * Cannot delete the last admin — refuses when *username*
205          is the sole entry with :data:`Role.ADMIN`. Otherwise a
206          botched cleanup would leave the install with no admin
207          user, which no future user-CRUD flow could recover from.
208        """
209        if username == current_username:
210            raise ValueError("cannot delete your own account")
211        target = self._users.by_username(username)
212        if target is None:
213            raise LookupError(f"no user {username!r}")
214        if target.role is Role.ADMIN:
215            admins = [u for u in self._users.users if u.role is Role.ADMIN]
216            if len(admins) <= 1:
217                raise ValueError(
218                    "cannot delete the last admin — promote someone else first",
219                )
220        self._users.users = [
221            u for u in self._users.users if u.username != username
222        ]
223        self._persist()
224
225    def set_password(self, username: str, new_password: str) -> None:
226        """Update *username*'s password hash. Existing session
227        cookies stay valid (they encode username only) — a
228        password change without a forced-logout is the JMRI
229        Java convention and matches "remember me" behaviour."""
230        user = self._users.by_username(username)
231        if user is None:
232            raise LookupError(f"no user {username!r}")
233        user.password_hash = hash_password(new_password)
234        self._persist()
235
236    def set_role(
237        self, username: str, new_role: Role, *, current_username: str,
238    ) -> None:
239        """Change *username*'s role. Same last-admin guard as
240        :meth:`remove_user` — demoting the only admin would
241        strand the install."""
242        if username == current_username and new_role is not Role.ADMIN:
243            raise ValueError("cannot demote your own account")
244        user = self._users.by_username(username)
245        if user is None:
246            raise LookupError(f"no user {username!r}")
247        if user.role is Role.ADMIN and new_role is not Role.ADMIN:
248            admins = [u for u in self._users.users if u.role is Role.ADMIN]
249            if len(admins) <= 1:
250                raise ValueError(
251                    "cannot demote the last admin — promote someone else first",
252                )
253        user.role = new_role
254        self._persist()
255
256    def regenerate_token(self, username: str) -> str:
257        """Mint a new token for *username*, replacing the old.
258        Returns the new value. Any mobile client using the old
259        token is immediately locked out — that's the point of
260        this endpoint (a leaked token gets cycled)."""
261        user = self._users.by_username(username)
262        if user is None:
263            raise LookupError(f"no user {username!r}")
264        user.token = secrets.token_urlsafe(32)
265        self._persist()
266        return user.token
267
268    def _persist(self) -> None:
269        """Write the current users file to disk. Called after
270        every mutation so an operator hand-editing users.yaml
271        alongside the running server sees a coherent file."""
272        save_users_file(self._users_path, self._users)

def save_users_file( path: str | pathlib._local.Path, file: UsersFile) -> None: 
 91def save_users_file(path: str | pathlib.Path, file: UsersFile) -> None:
 92    """Atomically write *file* to *path* with mode 0600 on POSIX.
 93
 94    Uses temp file + rename so a crashed save can't leave the
 95    file half-written. On Windows, the ``chmod`` is a no-op (the
 96    OS ACL isn't representable through the same mode bits); the
 97    settings root should be user-only anyway.
 98    """
 99    p = pathlib.Path(path)
100    p.parent.mkdir(parents=True, exist_ok=True)
101    # ``mode="json"`` converts the ``Role`` enum to its string value —
102    # pyyaml's safe_dump can't represent StrEnum instances directly,
103    # and even if it could, we want the on-disk shape to be plain
104    # strings so a human editor doesn't see ``!!python/object:``
105    # references.
106    payload = {"users": [u.model_dump(mode="json") for u in file.users]}
107    fd, tmp = tempfile.mkstemp(
108        prefix=".users.yaml.", dir=str(p.parent),
109    )
110    try:
111        with os.fdopen(fd, "w", encoding="utf-8") as f:
112            yaml.safe_dump(payload, f, sort_keys=False)
113        # Best-effort mode 0600 — matters on multi-user hosts, no-op
114        # on Windows where the ACL model differs.
115        try:
116            os.chmod(tmp, 0o600)
117        except (OSError, PermissionError) as e:  # pragma: no cover
118            _log.debug("chmod on %s ignored: %s", tmp, e)
119        os.replace(tmp, p)
120    except Exception:
121        try:
122            os.unlink(tmp)
123        except OSError:
124            pass
125        raise

38def verify_password(password: str, password_hash: str) -> bool:
39    """Constant-time check of *password* against *password_hash*.
