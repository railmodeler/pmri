jmri_server.topic_help

Human help for each JSON API topic.

Rendered as an accordion under the "Topics available on the JSON API" card on the home tab. Each TopicHelp entry gets one badge on the home page that, when clicked, expands to show the summary, a methods table, and an example round-trip.

Content policy — every method entry needs at minimum a one-sentence description and either an example request or a note saying no data is required. Keep examples short: they're for "remind me what the shape is," not full protocol docs. Full protocol docs live in the generated pdoc reference under /docs/.

Adding a new service:

  • Register a new TopicHelp in TOPIC_HELP.
  • If you can't fill in an example immediately, use example_request=None — the UI hides the block cleanly.
  • The source field is a short pointer for developers reading the reference; usually jmri_server.services.<module>.<Class>.

 22from __future__ import annotations

 24from dataclasses import dataclass, field

 26
 27@dataclass(frozen=True, slots=True)
 28class MethodHelp:
 29    """One method (``list`` / ``get`` / ``put`` / ``post`` / ``delete``)."""
 30
 31    name: str
 32    description: str
 33    example_request: str | None = None
 34    example_response: str | None = None
 35
 36
 37@dataclass(frozen=True, slots=True)
 38class TopicHelp:
 39    """Everything the accordion needs to render one topic's help panel."""
 40
 41    topic: str
 42    summary: str
 43    methods: list[MethodHelp] = field(default_factory=list)
 44    # Short "read the code" pointer — module.ClassName, no leading dots.
 45    source: str = ""
 46    # Any topic that isn't in JMRI Java's original API gets ``True`` so
 47    # the UI can badge it as rewrite-specific. Keeps client-side JMRI
 48    # tooling from expecting it against a legacy server.
 49    rewrite_extension: bool = False

 52# ---- shared examples ---------------------------------------------------
 53
 54# Every bean list follows this shape, so keep one canonical example
 55# to avoid drift across the eight bean services.
 56_BEAN_LIST_EXAMPLE = '{"type": "turnout", "method": "list"}'
 57_BEAN_LIST_RESPONSE = (
 58    '{"type": "list", "data": [\n'
 59    '  {"name": "IT1", "userName": "Main", "state": 2, "inverted": false}\n'
 60    "]}"
 61)
 62_BEAN_GET_EXAMPLE = (
 63    '{"type": "turnout", "method": "get", "data": {"name": "IT1"}}'
 64)
 65_BEAN_POST_EXAMPLE = (
 66    '{"type": "turnout", "method": "post",\n'
 67    ' "data": {"name": "IT1", "state": 2}}'
 68)
 69
 70
 71def _bean_methods(kind: str) -> list[MethodHelp]:
 72    """The standard four-method surface every bean service exposes."""
 73    return [
 74        MethodHelp(
 75            "list", f"Every registered {kind} with its current state.",
 76            example_request=f'{{"type": "{kind}", "method": "list"}}',
 77            example_response=_BEAN_LIST_RESPONSE.replace("turnout", kind),
 78        ),
 79        MethodHelp(
 80            "get", f"One {kind}'s current state by system name.",
 81            example_request=_BEAN_GET_EXAMPLE.replace("turnout", kind),
 82        ),
 83        MethodHelp(
 84            "post", f"Change one {kind}'s state (throw/close, active/inactive).",
 85            example_request=_BEAN_POST_EXAMPLE.replace("turnout", kind),
 86        ),
 87        MethodHelp(
 88            "put", f"Create or update a {kind}. Idempotent — a repeated "
 89            "put with the same fields is a no-op.",
 90        ),
 91    ]
