132    async def _poll_one(self, aiu: int) -> None:
133        try:
134            reply = await self._tc.send_command(read_aiu(aiu))
135        except (TimeoutError, Exception) as e:  # noqa: BLE001
136            _log.warning("NCE AIU-poll failed for %d: %s", aiu, e)
137            return
138        if len(reply.data) < 2:
139            return
140        word = (reply.data[0] << 8) | reply.data[1]
141        prev = self._snapshots.get(aiu, ~word & 0xFFFF)  # force diff first time
142        changed = prev ^ word
143        for bit in range(14):                 # 14 inputs per AIU
144            if not (changed & (1 << bit)):
145                continue
146            sensor_address = (aiu - 1) * 14 + bit + 1
147            sensor = self._mgr.get_by_address(sensor_address)
148            if sensor is None:
149                continue
150            state = SensorState.ACTIVE if (word & (1 << bit)) else SensorState.INACTIVE
151            await sensor._set_known_state(state)
152        self._snapshots[aiu] = word

252        """Read the CS's fast-clock state.
253
254        Talks to command-station memory at ``CS_CLOCK_ADDR`` and decodes
255        the 16-byte reply into a :class:`HardwareClockState`. In 12h mode
256        the CS reports hours 1..12 with an AM/PM flag; we normalise to
257        24-hour before returning so the JSON/web layer never has to care.
258        """
259        cmd = nce_messages.read_clock()
260        reply = await self.traffic.send_command(cmd)
261        data = reply.data
262        if _log.isEnabledFor(logging.DEBUG):
263            _log.debug(
264                "NCE clock read bytes: %s",
265                " ".join(f"{b:02X}" for b in data),
266            )
267        scale = data[0]
268        seconds = data[2] & 0xFF
269        minutes = data[3] & 0xFF
270        hours_raw = data[4] & 0xFF
271        ampm_flag = data[5]              # ASCII 'A' (65) = AM, else PM
272        twenty_four = (data[6] & 0xFF) == 1
273        # In 12h mode, hours cycle 12,1..11,12,1..11 with AM flipping to PM
274        # at each 12. Normalise: AM 12 → 0, PM 12 → 12, PM 1..11 → 13..23.
275        if twenty_four:
276            hours = hours_raw % 24
277        else:
278            is_am = ampm_flag == ord("A")
279            h = hours_raw % 12               # 12 → 0
280            hours = h if is_am else h + 12
281        # Scale of 0 would divide-by-zero; treat as "unknown, don't touch".
282        rate = float(250 // scale) if scale else 1.0
283        # The CS's 16-byte poll has no clean "running" bit — byte [1] is a
284        # scale-prescaler tick that stays nonzero even when stopped, so
285        # decoding it as a run flag reports running=True for a stopped
286        # clock. JMRI's ``NceClockControl`` infers running by watching
287        # tick advancement across reads instead; we do the same:
288        # if this read's seconds byte differs from the last one, the CS
289        # advanced → running. If identical, → stopped. On the very first
290        # read we have no history and fall back to whatever the last
291        # explicit start/stop button intent was.
292        if self._last_clock_seconds is None:
293            running = self._last_clock_running
294        else:
295            running = seconds != self._last_clock_seconds
296        self._last_clock_seconds = seconds
297        self._last_clock_running = running
298        return HardwareClockState(
299            hours=hours,
300            minutes=minutes,
301            seconds=seconds,
302            rate=rate,
303            running=running,
304            twenty_four_hour=twenty_four,
305        )

 30    """One command in flight at a time. All sends serialize through ``_lock``."""
 31
 32    # Fast-fail behaviour: once the CS has been silent for more than
 33    # ``SILENCE_THRESHOLD_S`` (matching the reachability monitor's
 34    # window plus a small margin so the two don't fight for control),
 35    # ``send_command`` refuses to touch the serial port and raises
 36    # ``TimeoutError`` immediately. Otherwise the browser's 1-Hz
 37    # fast-clock polls + the turnout monitor's 1.5 s poll cadence
 38    # queue up behind the ``_lock`` and each pays the full 500 ms
 39    # timeout — a home-page render or profile-load can stall for
 40    # 5–10 seconds even though we already know the CS is off.
 41    #
 42    # Every ``PROBE_INTERVAL_S`` one command IS let through to probe
 43    # whether the CS has come back. On success the timestamp gets
 44    # updated, silence drops below threshold, and normal operation
 45    # resumes.
 46    SILENCE_THRESHOLD_S = 4.0
 47    PROBE_INTERVAL_S = 5.0
 48
 49    def __init__(self, port: BytePort) -> None:
 50        self._port = port
 51        self._lock = asyncio.Lock()
 52        self._closed = False
 53        # Timestamp (``time.monotonic()`` scale) of the most recent
 54        # completed round-trip with the CS. ``None`` before any
 55        # command has been sent. Read by :class:`NceConnection`'s
 56        # reachability monitor to flip the CS-power state; write-only
 57        # here — updates on every success in :meth:`send_command` and
 58        # in :meth:`consume_trailer` when the trailer arrives.
 59        self.last_success_at: float | None = None
 60        # Fallback reference for the fast-fail silence calculation.
 61        # If the CS has been off since server start, ``last_success_at``
 62        # is always ``None`` and would defeat the short-circuit —
 63        # every command would still pay its own serial timeout. Using
 64        # ``max(start_time, last_success_at)`` means an always-off CS
 65        # is detected the moment startup grace elapses, and a CS that
 66        # went off after some traffic is measured from its last reply.
 67        self._start_time: float = time.monotonic()
 68        # Next moment at which a probe attempt is allowed while the CS
 69        # is known silent. See the class-level fast-fail comment.
 70        self._next_probe_at: float | None = None
 71
 72    async def aclose(self) -> None:
 73        self._closed = True
 74        await self._port.aclose()
 75
 76    async def send_command(self, cmd: NceCommand) -> NceReply:
 77        """Write *cmd*, await its reply with timeout, return the reply data.
 78
 79        Raises:
 80          - :class:`asyncio.TimeoutError` if the command station doesn't
 81            answer within ``cmd.timeout_s``;
 82          - :class:`NceCommandError` if the ACK byte indicates failure.
 83        """
 84        if self._closed:
 85            raise RuntimeError("traffic controller is closed")
 86        # Fast-fail if the CS has been silent long enough that the
 87        # reachability monitor has already flipped it to unreachable.
 88        # Without this, a browser page-load can queue ~10 commands (fast
 89        # clock + turnout monitor + user actions) that each pay 500 ms
 90        # of serial timeout in series — 5–9 s of hang on GET / and
 91        # POST /profiles/load. Let one probe through every
 92        # ``PROBE_INTERVAL_S`` so recovery is detected.
 93        #
 94        # ``reference`` is the later of first-open and last successful
 95        # round-trip. If the CS has been off since startup,
 96        # ``last_success_at`` is ``None`` — measuring silence from
 97        # ``_start_time`` still trips the short-circuit once startup
 98        # grace elapses.
 99        now = time.monotonic()
100        reference = (
101            self.last_success_at
102            if self.last_success_at is not None
103            else self._start_time
104        )
