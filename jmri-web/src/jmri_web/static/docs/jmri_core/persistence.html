jmri_core.persistence

XML persistence — load/save of JMRI configuration files.

Stage 1 shipped: per-loco locomotive-config round-trip (jmri_core.persistence.roster_xml).

Stage 2 adds:

  • jmri_core.persistence.profile — Profile and the standard ~/JMRI/ directory layout.
  • jmri_core.persistence.roster_index_xml — roster.xml index file (per roster.xsd).
  • jmri_core.persistence.decoder_xml — read-only decoder definition loader for the 881 xml/decoders/*.xml files.
  • jmri_core.persistence.operations_xml — semantic round-trip for all six Operations module file families.
  • jmri_core.persistence.panel_xml — envelope-level round-trip for PanelPro / LayoutEditor files (full element modeling is Stage 11).

 60class Profile:
 61    """One JMRI railroad profile.
 62
 63    Lives at ``<root>/<name>`` (e.g. ``~/JMRI/DRGW.jmri``). The bare name
 64    is whatever appears in ``profiles.xml`` — JMRI conventionally
 65    suffixes it with ``.jmri`` but we don't enforce that.
 66    """
 67
 68    root: pathlib.Path
 69    name: str = "default"
 70
 71    @classmethod
 72    def default(cls) -> Profile:
 73        """Construct using the platform-default settings directory."""
 74        return cls(root=default_settings_dir())
 75
 76    # ----- known paths ----------------------------------------------------
 77
 78    @property
 79    def profile_dir(self) -> pathlib.Path:
 80        """``<root>/<name>`` — the profile's root directory."""
 81        return self.root / self.name
 82
 83    @property
 84    def config_file(self) -> pathlib.Path:
 85        """The bean-tables file. Prefers ``PanelPro.xml`` (JMRI 5.x) if
 86        either it exists already; otherwise falls back to the older
 87        ``Configuration.xml``. Brand-new profiles get ``PanelPro.xml``.
 88        """
 89        panel = self.profile_dir / "PanelPro.xml"
 90        legacy = self.profile_dir / "Configuration.xml"
 91        if panel.exists():
 92            return panel
 93        if legacy.exists():
 94            return legacy
 95        return panel
 96
 97    @property
 98    def roster_dir(self) -> pathlib.Path:
 99        """Where per-loco ``locomotive-config`` XML files live."""
100        return self.profile_dir / "roster"
101
102    @property
103    def roster_index(self) -> pathlib.Path:
104        """Roster index — ``<profile>/roster.xml``, alongside the roster dir."""
105        return self.profile_dir / "roster.xml"
106
107    @property
108    def rosters_cv_dir(self) -> pathlib.Path:
109        """Per-loco CV state files (programmer output). Loaded on demand."""
110        return self.profile_dir / "rosters"
111
112    @property
113    def throttle_dir(self) -> pathlib.Path:
114        return self.profile_dir / "throttle"
115
116    @property
117    def signal_dir(self) -> pathlib.Path:
118        """Per-profile signal-mast definitions."""
119        return self.profile_dir / "signal"
120
121    @property
122    def programmers_dir(self) -> pathlib.Path:
123        return self.profile_dir / "programmers"
124
125    @property
126    def jython_dir(self) -> pathlib.Path:
127        """Legacy autoload dir (JMRI Java's Jython 2.7 tree). Kept as a
128        model attribute so imports from an older rewrite install still
129        parse — the autoloader no longer scans it. Use :attr:`python_dir`
130        for CPython 3 scripts."""
131        return self.profile_dir / "jython"
132
133    @property
134    def python_dir(self) -> pathlib.Path:
135        """Per-profile CPython 3 script dir.
136
137        Autoload sequence (see
138        :func:`jmri_server.app._autolaunch_profile_scripts`):
139
140        1. ``<settings-root>/python/`` — cross-profile shared scripts.
141        2. ``<profile>/python/`` — per-profile scripts.
142
143        Files ending in ``.py`` at the top level run in alphabetical
144        order; files starting with ``_`` are treated as private
145        modules that other scripts may import but aren't run standalone.
146        Missing directories are silently ignored — running without
147        either dir is a valid setup for a bare install.
148        """
149        return self.profile_dir / "python"
150
151    @property
152    def shared_python_dir(self) -> pathlib.Path:
153        """Cross-profile CPython 3 script dir at

155        autoload sequence."""
156        return self.root / "python"
157
158    @property
159    def operations_dir(self) -> pathlib.Path:
160        return self.profile_dir / "operations"
161
162    @property
163    def panel_dir(self) -> pathlib.Path:
164        return self.profile_dir / "panels"
165
166    @property
167    def logixng_dir(self) -> pathlib.Path:
168        """Where LogixNG programs are stored as YAML files.
169
170        Each program lives at ``<profile>/logixng/<system_name>.yaml``
171        so a profile switch swaps the whole automation set atomically.
172        Directory is created on first save; loads treat a missing dir
173        as "no programs" rather than an error.
174        """
175        return self.profile_dir / "logixng"
176
177    @property
178    def log_dir(self) -> pathlib.Path:
179        return self.root / "log"
180
181    @property
182    def preferences_dir(self) -> pathlib.Path:
183        return self.profile_dir / "profile"
184
185    @property
186    def signal_systems_dir(self) -> pathlib.Path:
187        """User-defined signal systems shared across profiles."""
188        return self.root / "resources" / "signals"
189
190    # ----- helpers ---------------------------------------------------------
191
192    def ensure(self) -> None:
193        """Create every standard per-profile subdirectory if missing.
194
195        Idempotent. Does not touch the settings-root level directories
196        (those are shared across profiles).
197        """
198        for d in (
199            self.root,
200            self.profile_dir,
201            self.roster_dir,
202            self.rosters_cv_dir,
203            self.operations_dir,
204            self.panel_dir,
205            self.throttle_dir,
206            self.signal_dir,
207            self.log_dir,
208            self.signal_systems_dir,
209        ):
210            d.mkdir(parents=True, exist_ok=True)
211
212    def __repr__(self) -> str:
213        return f"Profile(name={self.name!r}, root={str(self.root)!r})"

138def save_cv_values_into_locomotive_config(
139    path: str | pathlib.Path,
140    cv_values: dict[str, int],
141    *,
142    loco_id: str | None = None,
143) -> None:
144    """Update or insert a ``<values>`` block inside a ``locomotive-config``
145    file, preserving every other element.
146
147    * If *path* doesn't exist, does nothing (caller is expected to have
148      created the roster entry first via the roster service).
149    * If *cv_values* is empty, an existing ``<values>`` block is removed
150      entirely — the file still round-trips through the roster loader.
151    * When the file lists multiple locomotives (rare), *loco_id* picks
152      which one gets the update; otherwise the first is used.
153
154    Writes are atomic (tmp + rename) so a crash mid-write can't corrupt
155    the roster entry.
156    """
157    p = pathlib.Path(path)
158    if not p.exists():
159        return
160    parser = etree.XMLParser(
161        remove_blank_text=False,
162        resolve_entities=False,
163        load_dtd=False,
164        no_network=True,
165    )
166    tree = etree.parse(str(p), parser=parser)
167    root = tree.getroot()
168    if root.tag != "locomotive-config":
169        return
170    target_loco = None
171    for loco in root.iter("locomotive"):
172        if loco_id is None or loco.get("id", "") == loco_id:
173            target_loco = loco
174            break
175    if target_loco is None:
176        return
177
178    # Update in place: preserve any existing ``<values>`` sub-elements
179    # that AREN'T our per-CV entries. In particular JMRI Java writes a
180    # sibling ``<decoderDef>`` block inside ``<values>`` that tracks
181    # per-variable human values; wiping the whole ``<values>`` block
182    # (as an earlier revision did) destroys that data. Same story for
183    # any other unrecognised child element we don't own.
184    values_el = target_loco.find("values")
185    if values_el is None:
186        if not cv_values:
187            return
188        values_el = etree.SubElement(target_loco, "values")
189    else:
190        # Remove existing per-CV entries so we don't stack duplicates.
191        # Accept both element shapes we've written across versions.
192        for entry in list(values_el):
193            if isinstance(entry.tag, str) and entry.tag in ("CVvalue", "CV"):
194                values_el.remove(entry)
195
196    def _key(name: str) -> tuple[int, str]:
197        try:
198            return (0, f"{int(name):05d}")
199        except ValueError:
200            return (1, name)
201
202    if cv_values:
203        # Emit ``<CVvalue>`` — that's the tag name JMRI Java writes and
204        # reads. Interop-safe: a file we write will round-trip through
205        # JMRI Java, and a file JMRI Java wrote (with ``<CVvalue>``)
206        # gets picked up by our reader too.
207        #
208        # lxml's ``pretty_print`` only formats truly-new subtrees; when
209        # we append into an already-parsed element with existing
210        # whitespace nodes it silently packs siblings onto one line.
211        # Explicitly set ``.tail`` on each new entry to a newline +
212        # indent so the file stays hand-editable and diff-friendly.
213        # The last entry gets the closing-tag indent so ``</values>``
214        # lines up under its parent.
215        parent_depth = 0
216        node = values_el
217        while node is not None and node.tag != "locomotive-config":
218            parent_depth += 1
219            node = node.getparent()
220        indent_unit = "  "
221        entry_indent = "\n" + indent_unit * (parent_depth + 1)
222        close_indent = "\n" + indent_unit * parent_depth
223        sorted_names = sorted(cv_values, key=_key)
224        # Make sure the last-existing child's tail (if any) leads into
225        # the first new entry cleanly.
226        if len(values_el) > 0:
227            values_el[-1].tail = entry_indent
228        for i, name in enumerate(sorted_names):
229            el = etree.SubElement(values_el, "CVvalue", attrib={
230                "name": name,
231                "value": str(cv_values[name]),
232            })
233            el.tail = (
234                close_indent if i == len(sorted_names) - 1 else entry_indent
235            )
236    elif len(values_el) == 0:
237        # No per-CV entries left and no other children (decoderDef etc.)
238        # → drop the empty ``<values>`` element rather than leaving
239        # dead XML behind.
240        target_loco.remove(values_el)
241
242    tmp = p.with_suffix(p.suffix + ".tmp")
243    tree.write(
244        str(tmp),
245        xml_declaration=True,
246        encoding="utf-8",
247        pretty_print=True,
248        standalone=False,
249    )
250    tmp.replace(p)

119        """Turn a JMRI mask string into a structured mask.
120
121        Empty / all-``X`` / missing masks give a full-byte mask (shift=0,
122        width=8) — that's what JMRI treats as "no mask" and it means the

153        """

1552
