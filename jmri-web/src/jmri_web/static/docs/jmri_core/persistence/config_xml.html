jmri_core.persistence.config_xml

<layout-config> XML — typed sections + opaque pass-through.

The JMRI 5.x bean-table file (PanelPro.xml on real installs; Configuration.xml on the older shape) holds every persisted named bean plus panels, blocks, LogixNG, etc. Modelling every section is Stage 11 work. For alpha we ship typed load/save for the sections operators most commonly edit through the web UI:

  • <turnouts> — system name, user name, feedback, inverted, automate, comment
  • <sensors> — system name, user name, inverted, comment
  • <lights> — system name, user name, comment

Every other top-level section (<signalheads>, <memories>, <routes>, <blocks>, <paneleditor>, LogixNG, etc.) is preserved verbatim as raw lxml.

XML shapes

JMRI 5.x writes bean rows with <systemName> (and optional <userName> / <comment>) as child elements, with the remaining bean attributes on the parent::

<turnout feedback="MONITORING" inverted="false" automate="Off">
  <systemName>NT40</systemName>
  <userName>Durango Staging turnout 1</userName>
  <comment>Optional notes</comment>
</turnout>

Older JMRI installs (pre-5.x) wrote everything as parent attributes::

<turnout systemName="NT40" userName="..." feedback="MONITORING"/>

This module reads both forms but always writes the 5.x form for parity with current JMRI installs. Unknown attributes on each bean row are preserved verbatim through TurnoutRow.extra_attribs and friends.

 45# JMRI's manager-class string format — used as the ``class`` attribute on
 46# each `<turnouts>` / `<sensors>` etc. element. JMRI dispatches load/save
 47# handlers based on this string, and its ``layout-*.xsd`` schema requires
 48# it to match ``[a-zA-Z0-9.]*[a-z]`` (last char lowercase). The historical
 49# abstract names like ``AbstractTurnoutManagerConfigXML`` end in uppercase
 50# ``L`` and are schema-invalid — files with them fail to open in JMRI Java.
 51
 52# Default fallback classes for prefixes we don't know how to map. These
 53# are the Internal (concrete) config-XML classes, both real and
 54# schema-valid, so a file we emit stays loadable in JMRI Java even if we
 55# don't recognise the connection driver.
 56_DEFAULT_TURNOUT_MGR_CLASS = "jmri.jmrix.internal.configurexml.InternalTurnoutManagerXml"
 57_DEFAULT_SENSOR_MGR_CLASS  = "jmri.jmrix.internal.configurexml.InternalSensorManagerXml"
 58_DEFAULT_LIGHT_MGR_CLASS   = "jmri.jmrix.internal.configurexml.InternalLightManagerXml"

182@dataclass
183class ConfigFile:
184    """A parsed ``<layout-config>`` PanelPro.xml / Configuration.xml.
185
186    Typed sections (``turnout_sections``, ``sensor_sections``,
187    ``light_sections``) are lists because JMRI emits one element per
188    connection prefix. Other top-level elements (signal heads, memories,
189    routes, blocks, panels, LogixNG, etc.) live in ``preserved`` and are
190    written back verbatim.
191    """
192
193    root_attribs: dict[str, str] = field(default_factory=dict)
194    doctype: str | None = None
195    turnout_sections: list[TurnoutSection] = field(default_factory=list)
196    sensor_sections: list[SensorSection] = field(default_factory=list)
197    light_sections: list[LightSection] = field(default_factory=list)
198    # Other top-level elements we don't model, kept verbatim:
199    preserved: list[etree._Element] = field(default_factory=list)
200
201    # ----- accessors -------------------------------------------------------

203    def all_turnouts(self) -> Iterable[TurnoutRow]:
204        for section in self.turnout_sections:
205            yield from section.rows
206

208        for section in self.sensor_sections:
209            yield from section.rows
210
211    def all_lights(self) -> Iterable[LightRow]:

219def _parse_bool(value: str | None, default: bool = False) -> bool:
220    """JMRI booleans are ``"true"``/``"false"`` or sometimes ``"yes"``/``"no"``."""
221    if value is None:
222        return default
223    return value.strip().lower() in ("true", "yes", "1")

226def _child_text(elem: etree._Element, tag: str) -> str | None:
227    """Return the trimmed text of the first ``<tag>`` child, if any."""
228    child = elem.find(tag)
229    if child is None or child.text is None:
230        return None
231    text = child.text.strip()
232    return text or None

330def load_config_bytes(data: bytes) -> ConfigFile:
331    """Parse a ``<layout-config>`` PanelPro/Configuration file from bytes.
332
333    ``remove_blank_text=True`` strips the whitespace-only text nodes
334    JMRI writes between siblings so lxml's ``pretty_print`` on save
335    produces cleanly indented output. Without this, freshly-added
336    ``<turnouts>`` / ``<sensors>`` sections come out on one gigantic
337    line — unreadable when a user tries to hand-edit the file. Safe
338    for us because the file is element-only (no mixed content that
339    would care about internal whitespace)."""
340    parser = etree.XMLParser(
341        remove_blank_text=True,
342        resolve_entities=False,
343        load_dtd=False,
344        no_network=True,
345    )
346    root = etree.fromstring(data, parser=parser)
347    if root.tag != "layout-config":
348        raise ValueError(
349            f"expected <layout-config> root, got <{root.tag}>"
350        )
351    tree = root.getroottree()
352    doctype: str | None = getattr(tree.docinfo, "doctype", None) or None
353
354    config = ConfigFile(
355        root_attribs={k: v for k, v in root.attrib.items()},
356        doctype=doctype,
357    )
358    for child in root:
359        if not isinstance(child.tag, str):
360            config.preserved.append(child)
361            continue
362        if child.tag == "turnouts":
363            config.turnout_sections.append(_parse_turnout_section(child))
364        elif child.tag == "sensors":
365            config.sensor_sections.append(_parse_sensor_section(child))
366        elif child.tag == "lights":
367            config.light_sections.append(_parse_light_section(child))
368        else:
369            config.preserved.append(child)
370    return config
371
372
373def load_config(path: str | pathlib.Path) -> ConfigFile:
374    return load_config_bytes(pathlib.Path(path).read_bytes())

443def save_config_bytes(config: ConfigFile) -> bytes:
444    """Serialize a :class:`ConfigFile` back to XML bytes (JMRI 5.x shape)."""
445    root = etree.Element("layout-config", attrib=config.root_attribs)
446    for section in config.turnout_sections:
447        _write_turnout_section(root, section)
448    for section in config.sensor_sections:
449        _write_sensor_section(root, section)
450    for section in config.light_sections:
451        _write_light_section(root, section)
452    for elem in config.preserved:
453        root.append(elem)
454    common_kwargs: dict[str, object] = {
455        "xml_declaration": True,
456        "encoding": "utf-8",
457        "pretty_print": True,
458        "standalone": False,
459    }
460    if config.doctype:
461        common_kwargs["doctype"] = config.doctype
462    return etree.tostring(root, **common_kwargs)
463

465def save_config(path: str | pathlib.Path, config: ConfigFile) -> None:
466    pathlib.Path(path).write_bytes(save_config_bytes(config))

def load_config_bytes(data: bytes) -> ConfigFile: 
331def load_config_bytes(data: bytes) -> ConfigFile:
332    """Parse a ``<layout-config>`` PanelPro/Configuration file from bytes.
333
334    ``remove_blank_text=True`` strips the whitespace-only text nodes
335    JMRI writes between siblings so lxml's ``pretty_print`` on save
336    produces cleanly indented output. Without this, freshly-added
337    ``<turnouts>`` / ``<sensors>`` sections come out on one gigantic
338    line — unreadable when a user tries to hand-edit the file. Safe
339    for us because the file is element-only (no mixed content that
340    would care about internal whitespace)."""
341    parser = etree.XMLParser(
342        remove_blank_text=True,
343        resolve_entities=False,
344        load_dtd=False,
345        no_network=True,
346    )
347    root = etree.fromstring(data, parser=parser)
348    if root.tag != "layout-config":
349        raise ValueError(
350            f"expected <layout-config> root, got <{root.tag}>"
351        )
352    tree = root.getroottree()
353    doctype: str | None = getattr(tree.docinfo, "doctype", None) or None
354
355    config = ConfigFile(
356        root_attribs={k: v for k, v in root.attrib.items()},
357        doctype=doctype,
358    )
359    for child in root:
360        if not isinstance(child.tag, str):
361            config.preserved.append(child)
362            continue
363        if child.tag == "turnouts":
364            config.turnout_sections.append(_parse_turnout_section(child))
365        elif child.tag == "sensors":
366            config.sensor_sections.append(_parse_sensor_section(child))
367        elif child.tag == "lights":
368            config.light_sections.append(_parse_light_section(child))
369        else:
370            config.preserved.append(child)
371    return config
