 48class ConsistManager:
 49    """Track a set of consists and fan throttle commands to their members.
 50
 51    The base implementation is the **software consist** manager: it
 52    stores consists in a dict keyed by the group's DCC address and
 53    executes group speed/direction/function commands by looping over
 54    every member's :class:`Throttle`. Drivers that support hardware
 55    consisting subclass this and override :meth:`_after_create` and
 56    :meth:`_before_delete` to install/remove the wire-level group;
 57    they typically leave the fan-out helpers alone since the base's
 58    per-loco commands are still valid even when the CS is grouping.
 59
 60    ``created`` and ``deleted`` are :class:`Signal` fan-outs so the
 61    push side of the JSON server, the WiThrottle broadcast, and the
 62    web UI's live table can react without polling.
 63    """
 64
 65    default_kind: ConsistKind = "software"
 66
 67    def __init__(
 68        self, throttles: ThrottleManager, system_prefix: str,
 69        programmers: ProgrammerManager | None = None,
 70    ) -> None:
 71        if not system_prefix:
 72            raise ValueError("system_prefix must be a non-empty string")
 73        self._throttles = throttles
 74        # Ops-mode programmer manager, used by the ``advanced`` kind to
 75        # write CV19 on each member decoder as they join or leave. May
 76        # be ``None`` on non-hardware setups (tests, InternalConnection);
 77        # advanced consists degrade silently to a no-op in that case
 78        # rather than raising.
 79        self._programmers: ProgrammerManager | None = programmers
 80        self.system_prefix = system_prefix
 81        self._consists: dict[int, Consist] = {}
 82        # Runtime drive state per consist address. Not persisted — the
 83        # consist itself is pure config (membership + wire kind), and
 84        # loco throttles carry the authoritative state. But the UI
 85        # needs a per-consist "last commanded" value to render the
 86        # slider position and highlight the active direction button;
 87        # without it every drive command re-renders the card at the
 88        # hardcoded defaults and the user sees the slider snap back to
 89        # 0 on every drag-release. Reset on delete.
 90        self._speeds: dict[int, float] = {}
 91        self._directions: dict[int, ThrottleDirection] = {}
 92        self.created: Signal[Consist] = Signal()
 93        self.deleted: Signal[Consist] = Signal()
 94        self.updated: Signal[Consist] = Signal()

 96    def rebind_throttles(self, throttles: ThrottleManager) -> None:

189    async def add_member(
190        self,
191        consist_address: int,
192        member_address: int,
193        *,
194        long_address: bool = False,
195        direction_normal: bool = True,
196    ) -> ConsistLoco:
197        """Add *member_address* to the consist at *consist_address*.
198
199        The member is appended at the end. If a loco with the same
200        address already exists in the consist, its direction flag is
201        updated in place and the existing :class:`ConsistLoco` is
202        returned. Fires :attr:`updated` on the consist so subscribers
203        can react to membership changes.
204        """
205        consist = self._consists.get(consist_address)
206        if consist is None:
207            raise KeyError(f"no consist at address {consist_address}")
208        for loco in consist.locos:
209            if loco.address == member_address:
210                loco.direction_normal = direction_normal
211                await self._after_membership_change(consist)
212                if consist.kind == "advanced":

214                        member_address, long_address=loco.long_address,
215                        consist_address=consist_address,
216                        direction_normal=direction_normal,
217                    )
218                await self.updated.emit(consist)
219                return loco
220        loco = ConsistLoco(
221            address=member_address,
222            long_address=long_address,
223            direction_normal=direction_normal,
224            position=len(consist.locos),
225        )
226        consist.locos.append(loco)
227        await self._after_membership_change(consist)
228        if consist.kind == "advanced":
229            await self._program_cv19(
230                member_address, long_address=long_address,
231                consist_address=consist_address,
232                direction_normal=direction_normal,

237    async def _program_cv19(
238        self,
239        member_address: int,
240        *,
241        long_address: bool,
242        consist_address: int,
243        direction_normal: bool,
244    ) -> None:
245        """Write CV19 on *member_address*'s decoder for an advanced
246        consist.
247
248        Per NMRA S-9.2.2, CV19 holds a 7-bit consist address in bits
249        0..6 and the direction flag in bit 7. Bit 7 set means the loco
250        runs in the reverse direction relative to the consist — i.e.
251        ``direction_normal=False``. A value of 0 in the low 7 bits
252        disables the consist for that decoder.
253        """
254        prog_mgr = self._programmers
255        if prog_mgr is None:
256            _log.debug(
257                "advanced consist %d: no programmer manager, skipping "
258                "CV19 write for member %d",
259                consist_address, member_address,
260            )
261            return
262        prog = prog_mgr.ops_programmer(
263            member_address, long_address=long_address,
264        )
265        if prog is None:
266            _log.debug(
267                "advanced consist %d: driver returned no ops "
268                "programmer for member %d; CV19 not written",
269                consist_address, member_address,
270            )
271            return
272        value = consist_address & 0x7F
273        if not direction_normal:
274            value |= 0x80
275        try:
276            await prog.write_cv(19, value)
277        except Exception as exc:  # noqa: BLE001
278            # POM writes can fail (no ack, decoder not on the track,
279            # RailCom garble). Log and keep going — the consist itself
280            # remains configured; the user can re-add the member after
281            # putting the loco on the rails.
282            _log.warning(
283                "CV19 write failed for consist %d, member %d "
284                "(value=%d): %s",
285                consist_address, member_address, value, exc,
286            )

310    async def remove_member(
311        self, consist_address: int, member_address: int,
312    ) -> bool:
313        """Drop *member_address* from the consist. Returns True if found.
314
315        Re-numbers surviving members' ``position`` fields so they stay
316        contiguous — the ordering is used for lead-loco / rear-loco
317        semantics in DCC-EX and NCE displays.
318        """
319        consist = self._consists.get(consist_address)
320        if consist is None:
321            return False
322        removed = False
323        removed_long_address = False
324        surviving: list[ConsistLoco] = []
325        for loco in consist.locos:
326            if loco.address == member_address and not removed:
327                removed = True
328                # Capture before the loco leaves — the ops programmer
329                # for the CV19 clear is keyed by the decoder's real
330                # address form (short vs long).
331                removed_long_address = loco.long_address
332                continue
333            surviving.append(loco)
