jmri_loconet.connection

LocoNetConnection — the SystemConnection subclass.

Construct, configure, open(...): builds the appropriate LocoNetPort, starts the LocoNetTrafficController, wires up bean managers + power + throttle. On close() everything is torn down cleanly.

Settings dict: { "kind": "loopback" | "serial" | "tcp", # for serial: "port": "COM5", "baud": 16457, # for tcp: "host": "lbserver.local", "tcp_port": 1234, # for loopback (tests): "loopback_port": LoopbackPort, }

 52class LocoNetConnection(SystemConnection):
 53    """LocoNet hardware connection. Prefix ``L``, name ``LocoNet``."""

 56    prefix: ClassVar[str] = "L"

 58    def __init__(self) -> None:
 59        super().__init__()
 60        # These are filled in by open(). The base-class defaults stay in place
 61        # until then so the connection has well-formed managers if opened with
 62        # no settings (rare; mainly the test path).
 63        self._port: LocoNetPort | None = None
 64        self._tc: LocoNetTrafficController | None = None
 65        self._slots: SlotManager | None = None
 66        self._programmer: LnProgrammer | None = None
 67        self._cs_monitor_task: asyncio.Task[None] | None = None

 69    # ----- lifecycle --------------------------------------------------------
 70
 71    async def open(self, settings: ConnectionSettings | None = None) -> None:
 72        s = settings or {}
 73        self._port = await _build_port(s)
 74        self._tc = LocoNetTrafficController(self._port)
 75        await self._tc.start()
 76        self._slots = SlotManager(self._tc)
 77
 78        # Replace the base default managers with our LocoNet-aware ones.
 79        # ``LnTurnoutManager`` and friends are subclasses of the base managers,
 80        # but mypy's invariance in attribute typing means we have to ignore
 81        # the assignment — the runtime substitution is sound (Liskov).
 82        self.turnouts = LnTurnoutManager(self.prefix, tc=self._tc)   # type: ignore[assignment]
 83        self.sensors = LnSensorManager(self.prefix, tc=self._tc)     # type: ignore[assignment]
 84        throttle_mgr = LnThrottleManager(
 85            self.prefix, tc=self._tc, slot_manager=self._slots,
 86        )
 87        self.throttles = throttle_mgr
 88        # Rebind the consist manager onto the LocoNet-aware throttle
 89        # manager. Without this, consist fan-out drives placeholder
 90        # Throttle beans left over from base ``__init__``, and no
 91        # OPC_LOCO_SPD frames reach the wire.
 92        self.consists.rebind_throttles(self.throttles)
 93        throttle_mgr.start_refresh()
 94        self.consists = LnConsistManager(   # type: ignore[assignment]
 95            throttle_mgr, self.prefix, tc=self._tc, slots=self._slots,
 96        )
 97        self.power = LnPowerManager(self._tc)
 98        self._programmer = LnProgrammer(self._tc)
 99        self.programmers = LnProgrammerManager(self._programmer)
100        # Wire the advanced-consist CV19 write path to the LocoNet
101        # programmer manager. Without this rebind, ``advanced`` consists
102        # go through the base placeholder manager and CV19 never
103        # actually programs.
104        self.consists.rebind_programmers(self.programmers)
105
106        # CS reachability monitor. LocoNet CS emit OPC_IDLE and other
107        # frames continuously; 3s of complete silence is a strong
108        # signal the bus is dark (CS off or unplugged). Start
109        # pessimistic (base class default is optimistic True) so a
110        # home-page render during the monitor's startup grace shows
111        # "Off" until we've actually seen a frame, not a false "On".
112        self._cs_reachable = False
113        self._cs_monitor_task = asyncio.create_task(
114            self._cs_reachability_loop(), name="loconet-cs-reachability",
115        )
116
117        await super().open(settings)

119    async def _cs_reachability_loop(self) -> None:
120        """Flip ``_cs_reachable`` on the 3-second inbound-silence rule.
121
122        LocoNet's continuous chatter (CS heartbeat + slot poll traffic)
123        means any real silence past 3 seconds indicates the CS has
124        stopped talking. Any fresh inbound frame flips reachable back
125        to True immediately.
126        """
127        import contextlib  # noqa: PLC0415
128        try:
129            # 3-second grace so the first CS heartbeat can arrive.
130            await asyncio.sleep(3.0)
131            while True:
132                await asyncio.sleep(1.0)
133                if self._tc is None:
134                    return
135                last = self._tc.last_inbound_at
136                if last is None:
137                    await self._set_cs_reachable(False)
138                    continue
139                age = time.monotonic() - last
140                await self._set_cs_reachable(age <= 3.0)
141        except asyncio.CancelledError:
142            with contextlib.suppress(Exception):
143                return
144
145    async def close(self) -> None:
146        if isinstance(self.throttles, LnThrottleManager):
147            await self.throttles.stop_refresh()
148        if self._cs_monitor_task is not None:
149            self._cs_monitor_task.cancel()
150            import contextlib  # noqa: PLC0415
151            with contextlib.suppress(asyncio.CancelledError, Exception):
152                await self._cs_monitor_task
153            self._cs_monitor_task = None
154        if self._tc is not None:
155            await self._tc.aclose()
156            self._tc = None
157        self._port = None
158        await super().close()

176    async def read_hardware_clock(self):  # noqa: ANN201
177        """Read the CS fast clock from slot 123.
178
179        Sends ``OPC_RQ_SL_DATA(slot=123)`` and awaits the paired
180        ``OPC_SL_RD_DATA`` reply on the inbound signal. Returns a
181        :class:`~jmri_core.beans.fast_clock.HardwareClockState`.
182
183        The :class:`~jmri_server.services.fast_clock.TimeService`
184        prefers this method over the software bean whenever the
185        connection exposes it, so wiring this on ``LocoNetConnection``
186        automatically makes ``/json/time`` reflect the CS clock.
187        """
188        from jmri_loconet.clock import FAST_CLOCK_SLOT, decode_slot123  # noqa: PLC0415
189        from jmri_loconet.messages import OpcRqSlData, OpcSlRdData  # noqa: PLC0415
190
191        if self._tc is None:
192            raise RuntimeError("connection not open")
193        tc = self._tc
194        loop = asyncio.get_event_loop()
195        fut: asyncio.Future[OpcSlRdData] = loop.create_future()
196
197        async def on_inbound(msg) -> None:  # noqa: ANN001
198            if (
199                isinstance(msg, OpcSlRdData)
200                and msg.slot == FAST_CLOCK_SLOT
201                and not fut.done()
202            ):
203                fut.set_result(msg)
204
205        unsub = tc.inbound.connect(on_inbound)
206        try:
207            await tc.send(OpcRqSlData(slot=FAST_CLOCK_SLOT))
208            reply = await asyncio.wait_for(fut, timeout=2.0)
209        finally:
210            unsub()
211        return decode_slot123(reply)

213    async def _write_hardware_clock(self, state) -> None:  # noqa: ANN001
214        """Common write path — reads the current slot to preserve
215        non-clock fields, merges *state*, and sends the write."""
216        current = await self.read_hardware_clock()
217        # ``current`` gave us the CS's authoritative days / trk / snd
218        # bytes indirectly by reading the raw slot; we lose that in
219        # the decode. To keep this pass simple we send zeros for those
220        # fields — JMRI Java's LnClockControl does the same on write,
221        # relying on the CS to preserve day count internally.
222        from jmri_loconet.clock import build_slot123_write  # noqa: PLC0415
223
224        merged = state.__class__(
225            hours=state.hours if state.hours is not None else current.hours,
226            minutes=state.minutes if state.minutes is not None else current.minutes,
227            seconds=state.seconds if state.seconds is not None else current.seconds,
228            rate=state.rate if state.rate is not None else current.rate,
229            running=state.running if state.running is not None else current.running,
230            twenty_four_hour=True,
231        )
232        msg = build_slot123_write(merged)
233        if self._tc is None:
234            raise RuntimeError("connection not open")
235        await self._tc.send(msg)

308async def _build_port(settings: dict[str, Any]) -> LocoNetPort:
309    kind = settings.get("kind", "loopback")
310    if kind == "serial":
311        port = settings.get("port")
312        if not port:
313            raise ValueError("serial transport requires 'port'")
314        baud = int(settings.get("baud", 16457))
315        return await SerialPort.open(port=port, baud=baud)
316    if kind == "tcp":
317        host = settings.get("host")
318        tcp_port = settings.get("tcp_port")
319        if not host or tcp_port is None:
320            raise ValueError("tcp transport requires 'host' and 'tcp_port'")
321        return await TcpPort.open(host=host, port=int(tcp_port))
322    if kind == "loopback":
323        port = settings.get("loopback_port")
324        if not isinstance(port, LoopbackPort):
325            raise ValueError(
326                "loopback transport requires 'loopback_port' to be a LoopbackPort"
327            )
328        return port
329    raise ValueError(f"unknown LocoNet transport kind: {kind!r}")

179

181        ``OPC_SL_RD_DATA`` reply on the inbound signal. Returns a
182        :class:`~jmri_core.beans.fast_clock.HardwareClockState`.
183

185        prefers this method over the software bean whenever the
186        connection exposes it, so wiring this on ``LocoNetConnection``
187        automatically makes ``/json/time`` reflect the CS clock.
188        """
189        from jmri_loconet.clock import FAST_CLOCK_SLOT, decode_slot123  # noqa: PLC0415
190        from jmri_loconet.messages import OpcRqSlData, OpcSlRdData  # noqa: PLC0415
191
192        if self._tc is None:
193            raise RuntimeError("connection not open")
194        tc = self._tc
195        loop = asyncio.get_event_loop()
196        fut: asyncio.Future[OpcSlRdData] = loop.create_future()
197
198        async def on_inbound(msg) -> None:  # noqa: ANN001
199            if (
200                isinstance(msg, OpcSlRdData)
201                and msg.slot == FAST_CLOCK_SLOT
202                and not fut.done()
203            ):
204                fut.set_result(msg)
205
206        unsub = tc.inbound.connect(on_inbound)
207        try:

210        finally:
