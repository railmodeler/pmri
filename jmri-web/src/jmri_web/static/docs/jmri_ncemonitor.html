class BusReplay: 
 85class BusReplay:
 86    """Feed a captured stream into a :class:`LoopbackPort` for testing.
 87
 88    Construct with ``BusReplay(frames)`` (or via :meth:`from_file`); call
 89    :meth:`run` to start a background task that writes frames to the
 90    *write* side of a loopback pair. The *read* side is returned so callers
 91    can hand it to a :class:`CabBusMonitor`.
 92
 93    Timing modes:
 94
 95    * ``preserve_timing=False`` (default) — emit all frames immediately,
 96      one after the other. Good for unit tests.
 97    * ``preserve_timing=True`` — sleep between frames according to the
 98      captured timestamps. Useful when validating real-time behaviour.
 99
100    The replay always appends a trailing address byte after the last
101    frame so the monitor's "read until next address" loop completes.
102    """
103

105        self,
106        frames: Iterable[CapturedFrame],
107        *,
108        preserve_timing: bool = False,
109    ) -> None:
110        self._frames = list(frames)
111        self._preserve_timing = preserve_timing
112        self._task: asyncio.Task[None] | None = None
113
114    @classmethod
115    def from_file(
116        cls, path: str | pathlib.Path, *, preserve_timing: bool = False
117    ) -> BusReplay:
118        capture = BusCapture.read(path)
119        return cls(capture.frames, preserve_timing=preserve_timing)
120
121    async def play_into(self, write_port: LoopbackPort) -> None:
122        """Write every captured frame into *write_port*."""
123        last_t: float | None = None
124        for cf in self._frames:
125            if self._preserve_timing and last_t is not None:
126                delay = cf.t - last_t
127                if delay > 0:
128                    await asyncio.sleep(delay)
129            last_t = cf.t
130            if not cf.frame or not is_address_byte(cf.frame[0]):
131                _log.warning("dropping malformed captured frame: %s", cf.frame.hex())
132                continue
133            await write_port.write(cf.frame)
134        # Tail boundary byte so the monitor's "read until next addr" returns.
135        await write_port.write(bytes([0x80]))
136
137    def start(self, write_port: LoopbackPort) -> asyncio.Task[None]:
138        """Kick off the replay as a background task. Returns the task."""
139        self._task = asyncio.create_task(self.play_into(write_port), name="cabbus-replay")
140        return self._task

class CabBusMonitor: 
27class CabBusMonitor:
28    """Owns one :class:`BytePort` and a background read task.
29
30    Listeners connect to :attr:`inbound` and receive every decoded message
31    in the order it appeared on the bus.
32    """
33
34    def __init__(self, port: BytePort) -> None:
35        self._port = port
36        self._reader = PeekableReader(port.read_byte)
37        self._read_task: asyncio.Task[None] | None = None
38        self._closed = False
39        self.inbound: Signal[CabBusMessage] = Signal()
40
41    async def start(self) -> None:
42        if self._read_task is not None:
43            return
44        self._read_task = asyncio.create_task(self._read_loop(), name="cabbus-monitor")
45
46    async def aclose(self) -> None:
47        self._closed = True
48        if self._read_task is not None:
49            self._read_task.cancel()
50            import contextlib  # noqa: PLC0415
51            with contextlib.suppress(asyncio.CancelledError, Exception):
52                await self._read_task
53            self._read_task = None
54        await self._port.aclose()
55
56    async def _read_loop(self) -> None:
57        try:
58            while not self._closed:
59                try:
60                    frame = await read_message_peekable(self._reader)
61                except asyncio.CancelledError:
62                    return
63                except BodyOverflowError as e:
64                    # A stuck bus buffered past the cap. Log the
65                    # partial for diagnostics and drop it — the next
66                    # read_message call resyncs on the wire.
67                    _log.warning(
68                        "Cab Bus body overflow (%d bytes buffered): %s",
69                        len(e.partial), e.partial.hex(),
70                    )
71                    continue
72                except Exception as e:  # noqa: BLE001
73                    _log.warning("Cab Bus read error: %s", e)
74                    continue
75                try:
76                    msg = decode(frame)
77                except ValueError as e:
78                    _log.warning("Cab Bus bad frame %s: %s", frame.hex(), e)
79                    continue
80                await self.inbound.emit(msg)
81        except asyncio.CancelledError:
82            return
