jmri-scripting — user script host for the JMRI Python rewrite (Stage 9).

Loads JMRI-style Python scripts and runs them with a familiar shim of synchronous wrappers over the async core. Scripts written for JMRI's Jython runtime port with minimal changes (the most-used 200 attribute paths from the Stage 0 script-api ranking are covered).

68        """Run the full lifecycle: init → handle-loop → cleanup."""

115class ReplSession:
116    """Persistent Python interpreter bound to the JMRI shim.
117
118    One instance per server. Callers submit source fragments via
119    :meth:`submit`; state persists across submissions until
120    :meth:`reset` is called.
121    """
122
123    def __init__(self, shim: JmriShim) -> None:
124        self._shim = shim
125        self._lock = threading.Lock()
126        self._history: list[ReplEntry] = []
127        self._globals: dict[str, Any] = {}
128        # Build the initial namespace so the very first submission
129        # already has ``sensors`` / ``turnouts`` / etc. available.
130        self._install_globals()
131
132    def _install_globals(self) -> None:
133        """Populate ``self._globals`` with the shim's injected names
134        plus a couple of REPL-friendly built-ins."""
135        self._globals = {
136            "__name__": "__jmri_repl__",
137            "__builtins__": __builtins__,
138            **self._shim.globals(),
139        }
140        # A friendly reference to the shim itself, for scripts that
141        # want to reach through to the underlying async connection.
142        self._globals["shim"] = self._shim
143
144    # ---- public API ----------------------------------------------------
145
146    @property
147    def history(self) -> list[ReplEntry]:
148        """Snapshot of the history buffer, oldest first."""
149        with self._lock:
150            return list(self._history)
151
152    def reset(self) -> None:
153        """Rebuild the interpreter — drops every user variable,
154        import, and function definition. History is cleared too."""
155        with self._lock:
156            self._history.clear()
157            self._install_globals()
158
159    def submit(self, source: str) -> ReplResult:
160        """Execute *source* against the persistent interpreter.
161
162        Returns a :class:`ReplResult`. Bounded stdout capture means
163        a script that prints megabytes only surfaces the first
164        64 KB — a runaway loop can't OOM the server.
165
166        **Threading note**: this method blocks synchronously on the
167        current thread. It must NOT be called from the event-loop
168        thread if the submitted source might touch the shim, because
169        the shim's ``_block(loop, coro)`` schedules coroutines onto
170        that loop and waits for them — the loop can't advance if the
171        caller is holding its thread. Two supported patterns:
172
173        * **From an async coroutine on the loop thread**: wrap in
174          ``asyncio.to_thread(repl.submit, source)`` — that off-loads
175          the exec to a worker so the loop stays free.
176        * **From any other thread** (batch worker, sync test): call
177          directly.
178
179        :meth:`submit_async` is a thin async wrapper that does the
180        ``to_thread`` for you; web routes use it.
181        """
182        with self._lock:
183            # Incomplete? Client keeps accumulating.
184            if _needs_continuation(source):
185                entry = ReplEntry(source=source)
186                return ReplResult(entry=entry, incomplete=True)
187
188            entry = ReplEntry(source=source)
189            self._exec_one(source, entry)
190            self._push_entry(entry)
191            return ReplResult(entry=entry, incomplete=False)
192
193    async def submit_async(self, source: str) -> ReplResult:
194        """Async wrapper — safe to call from a coroutine running on
195        the same event loop the shim binds to. Off-loads the sync
196        exec to a worker thread so the loop stays free to service
197        shim coroutines."""
198        import asyncio  # noqa: PLC0415
199        return await asyncio.to_thread(self.submit, source)
200
201    # ---- internals ----------------------------------------------------
202
203    def _exec_one(self, source: str, entry: ReplEntry) -> None:
204        """Run *source* under stdout/stderr capture, filling *entry*.
205
206        Binds the shim to the current thread so
207        ``jmri.InstanceManager.getDefault(...)`` resolves — the shim
208        is thread-local (matches the batch runner's convention where
209        each script's worker thread gets its own bind).
210        """
211        from jmri_scripting import jython_shim  # noqa: PLC0415
212
213        stdout_buf = io.StringIO()
214        stderr_buf = io.StringIO()
215
216        jython_shim._bind_shim(self._shim)
217        try:
218            with (
219                contextlib.redirect_stdout(stdout_buf),
220                contextlib.redirect_stderr(stderr_buf),
221            ):
222                self._run_source_capturing(source)
223        except SystemExit:
224            entry.error = "SystemExit: refused to exit the server"
225        except BaseException:  # noqa: BLE001
226            entry.error = traceback.format_exc()
227        finally:
228            jython_shim._unbind_shim()
229
230        entry.stdout = _clip(stdout_buf.getvalue(), _MAX_STDOUT_PER_ENTRY)
231        entry.stderr = _clip(stderr_buf.getvalue(), _MAX_STDOUT_PER_ENTRY)
232
233    def _run_source_capturing(self, source: str) -> None:
234        """Two-pass compile: try as an expression to surface the
235        result via ``repr()``, else exec as statements.
236
237        Mirrors CPython's REPL: a bare expression at the prompt
238        prints its ``repr``; a statement produces no automatic
239        output. Storing the result on the *entry* is done by the
240        caller via ``__last_result__``.
241        """
242        # ``self._entry_being_run`` set by ``_exec_one`` so the
243        # ``compile...eval`` branch can stash the result. Threaded
244        # through a per-call attribute rather than a return value so
245        # the exception paths above can still see the partial
246        # entry state.
247        try:
248            expr_code = compile(source, "<repl>", "eval")
249        except SyntaxError:
250            expr_code = None
251
252        if expr_code is not None:
253            result = eval(expr_code, self._globals)  # noqa: S307
254            if result is not None:
255                # Print ``repr`` to stdout so the console rendering
256                # is one uniform place — no separate "result" cell.
257                print(repr(result))
258            return
259
260        # Statement form: compile + exec.
261        stmt_code = compile(source, "<repl>", "exec")
262        exec(stmt_code, self._globals)  # noqa: S102
263
264    def _push_entry(self, entry: ReplEntry) -> None:
265        self._history.append(entry)
266        overflow = len(self._history) - _MAX_HISTORY_ENTRIES
267        if overflow > 0:
268            del self._history[:overflow]

352        # Re-verify the dispatch streams are still installed. Some test
353        # runners (pytest) and long-running processes (uvicorn --reload)

355        # proxy — re-installing here wraps whatever is current so a

379    def _exec_script(
380        self,
381        name: str,
382        source: str,
383        path: pathlib.Path | None,
384        status: ScriptStatus,
385    ) -> None:
386        # Route this thread's stdout/stderr into the status buffers so
387        # ``print()`` and any raised traceback surface in the web UI
388        # rather than getting lost in the server's log stream. The
389        # dispatch streams were installed on runner construction; here
390        # we just register/unregister this thread's target.
391        self._stdout_proxy.register(status._stdout_buffer)
392        self._stderr_proxy.register(status._stderr_buffer)

403    def _exec_script_body(
404        self,
405        name: str,
406        source: str,
407        path: pathlib.Path | None,
408        status: ScriptStatus,
409    ) -> None:
410        globals_dict = self._shim.globals() | {
411            "AbstractAutomaton": AbstractAutomaton,
412            "__name__": "__jmri_script__",
413            "__file__": str(path) if path else f"<{name}>",
414        }
415        try:
416            code = compile(source, str(path) if path else f"<{name}>", "exec")
417            exec(code, globals_dict)  # noqa: S102 — script execution is the feature
418        except Exception:
419            status.error = traceback.format_exc()
420            status.finished = True
421            return
422
423        # After top-level execution, look for an automaton subclass to run.
424        automaton_cls = _find_automaton_class(globals_dict)
425        if automaton_cls is not None:
426            automaton = automaton_cls(name=name)
427            automaton.shim = self._shim
428            with self._lock:
429                status.automaton = automaton
430            try:
431                automaton.run()
432            except Exception:
433                status.error = traceback.format_exc()
434        # Or look for a `main(shim)` callable.
435        else:
436            main_fn = globals_dict.get("main")
437            if callable(main_fn):
438                try:
439                    sig = inspect.signature(main_fn)
440                    if len(sig.parameters) > 0:
441                        main_fn(self._shim)
442                    else:
443                        main_fn()
444                except Exception:
445                    status.error = traceback.format_exc()
446        status.finished = True

60                self._tick()
61                with contextlib.suppress(TimeoutError):
62                    await asyncio.wait_for(self._stop.wait(), timeout=self._interval)

71            try:
72                current[p] = p.stat().st_mtime
73            except FileNotFoundError:
74                continue
75

77        for p, mtime in current.items():
78            prev = self._mtimes.get(p)
79            if prev is None or mtime > prev + 1e-6:
80                _log.info("hot-reload: %s", p.name)
81                try:
82                    self._runner.run_file(p, name=p.stem)
83                except Exception as e:  # noqa: BLE001
84                    _log.warning("failed to re-run %s: %s", p, e)
85
86        # Deleted files: stop the running script.
87        for p in list(self._mtimes):
88            if p not in current:
89                _log.info("hot-reload: %s removed, stopping", p.name)
90                self._runner.stop(p.stem)
91
92        self._mtimes = current
