jmri_ncemonitor.messages

Typed Cab Bus observation messages.

The NCE Cab Bus has many message types; this module covers the ones a monitor user actually wants to see decoded. Anything else surfaces as CabBusUnknown carrying the raw bytes so callers don't lose it.

Addresses observed on the bus:

  • 0x00..0x3F (with high bit set on the wire → 0x80..0xBF) — cab IDs.
  • 0x60..0x7F — fast clock, accessory packets, master traffic.

185def decode(frame: bytes) -> CabBusMessage:
186    """Decode one Cab Bus frame into a typed message.
187
188    *frame* starts with the address byte (high bit set). Everything after
189    is body. Returns :class:`CabBusUnknown` for unrecognized frames.
190    """
191    if not frame or not is_address_byte(frame[0]):
192        raise ValueError("Cab Bus frame must start with an address byte")
193    addr = address_of(frame[0])
194    body = frame[1:]
195
196    # Cab range
197    if addr < 0x40:
198        if len(body) == 1:
199            return CabPoll(address=addr, command=body[0])
200        # JMRI's Cab Bus reference documents a fixed 9-byte cab
201        # response frame (5 state bytes + 3 pad + 1 checksum).
202        # Require the full length so short frames fall through to
203        # CabBusUnknown instead of silently succeeding on truncated
204        # data — the monitor operator wants to see truncation, not
205        # a plausible-looking half-decoded message.
206        if len(body) >= 9:
207            return CabResponse(
208                address=addr,
209                key=body[0],
210                speed=body[1] & 0x7F,
211                direction=(body[2] >> 0) & 0x01,
212                function_low=body[3] & 0x1F,
213                function_high=body[4] & 0x0F,
214            )
215
216    # Master-side traffic.
217    if addr == ADDR_FAST_CLOCK and len(body) >= 2:
218        return FastClockUpdate(address=addr, hour=body[0] & 0x7F, minute=body[1] & 0x7F)
219
220    if addr == ADDR_ACCESSORY and len(body) >= 3:
221        accy_addr = ((body[0] & 0x7F) << 7) | (body[1] & 0x7F)
222        closed = bool(body[2] & 0x01)
223        return AccessoryObserved(
224            address=addr, accessory_address=accy_addr + 1, closed=closed
225        )
226
227    if addr == ADDR_AIU_UPDATE and len(body) >= 3:
228        return AiuUpdate(
229            address=addr,
230            aiu_module=body[0] & 0x7F,
231            input_word=((body[1] & 0x7F) << 7) | (body[2] & 0x7F),
232        )
233
234    if addr == ADDR_THROTTLE_CMD and len(body) >= 4:
235        return ThrottleObserved(
236            address=addr,
237            dcc_address=((body[0] & 0x7F) << 7) | (body[1] & 0x7F),
238            speed=body[2] & 0x7F,
239            forward=bool(body[3] & 0x40),
240            functions=body[3] & 0x1F,
241        )
242
243    if addr == ADDR_PROG_RESP and len(body) >= 4:
244        return ProgrammingResponse(
245            address=addr,
246            cv=((body[0] & 0x7F) << 7) | (body[1] & 0x7F),
247            value=body[2] & 0xFF,
248            operation=body[3] & 0x03,
249        )
250
251    if addr == ADDR_CONSIST_CMD and len(body) >= 4:
252        return ConsistObserved(
253            address=addr,
254            consist_number=body[0] & 0x7F,
255            loco_address=((body[1] & 0x7F) << 7) | (body[2] & 0x7F),
256            operation=body[3] & 0x03,
257        )
258
259    if addr == ADDR_MACRO_CMD and len(body) >= 1:
260        return MacroObserved(
261            address=addr,
262            macro_number=body[0] & 0x7F,
263        )
264
265    if addr == ADDR_MASTER_ACK and len(body) >= 1:
266        return MasterAck(
267            address=addr,
268            code=body[0] & 0x7F,
269        )
270
271    return CabBusUnknown(address=addr, data=body)

274def encode(msg: CabBusMessage) -> bytes:
275    """Inverse of :func:`decode` — used only by tests / synthetic playback.
276
277    The real monitor never writes to the bus (it's passive); we provide
278    ``encode`` to construct test fixtures that feed the monitor through a
279    loopback transport.
280    """
281    if isinstance(msg, CabPoll):
282        return bytes([0x80 | msg.address, msg.command])
283    if isinstance(msg, CabResponse):
284        # 9-byte body: 5 state bytes + 3 pad + 1 checksum. The
285        # checksum is the XOR of the 5 state bytes on the wire — we
286        # emit it here so encode() → decode() round-trips against
287        # the tightened 9-byte length check in ``decode``.
288        state = (
289            msg.key & 0x7F,
290            msg.speed & 0x7F,
291            msg.direction & 0x7F,
292            msg.function_low & 0x7F,
293            msg.function_high & 0x7F,
294        )
295        checksum = 0
296        for b in state:
297            checksum ^= b
298        return bytes([
299            0x80 | msg.address,
300            *state,
301            0x00, 0x00, 0x00,
302            checksum & 0x7F,
303        ])
304    if isinstance(msg, FastClockUpdate):
305        return bytes([0x80 | ADDR_FAST_CLOCK, msg.hour & 0x7F, msg.minute & 0x7F])
306    if isinstance(msg, AccessoryObserved):
307        a = msg.accessory_address - 1
308        return bytes([
309            0x80 | ADDR_ACCESSORY,
310            (a >> 7) & 0x7F,
311            a & 0x7F,
312            0x01 if msg.closed else 0x00,
313        ])
314    if isinstance(msg, AiuUpdate):
315        return bytes([
316            0x80 | ADDR_AIU_UPDATE,
317            msg.aiu_module & 0x7F,
318            (msg.input_word >> 7) & 0x7F,
319            msg.input_word & 0x7F,
320        ])
321    if isinstance(msg, ThrottleObserved):
322        return bytes([
323            0x80 | ADDR_THROTTLE_CMD,
324            (msg.dcc_address >> 7) & 0x7F,
325            msg.dcc_address & 0x7F,
326            msg.speed & 0x7F,
327            (msg.functions & 0x1F) | (0x40 if msg.forward else 0),
328        ])
329    if isinstance(msg, ProgrammingResponse):
330        return bytes([
331            0x80 | ADDR_PROG_RESP,
332            (msg.cv >> 7) & 0x7F,
333            msg.cv & 0x7F,
334            msg.value & 0x7F,
335            msg.operation & 0x7F,
336        ])
337    if isinstance(msg, ConsistObserved):
338        return bytes([
339            0x80 | ADDR_CONSIST_CMD,
340            msg.consist_number & 0x7F,
341            (msg.loco_address >> 7) & 0x7F,
342            msg.loco_address & 0x7F,
343            msg.operation & 0x7F,
344        ])
345    if isinstance(msg, MacroObserved):
346        return bytes([
347            0x80 | ADDR_MACRO_CMD,
348            msg.macro_number & 0x7F,
349        ])
350    if isinstance(msg, MasterAck):
351        return bytes([
352            0x80 | ADDR_MASTER_ACK,
353            msg.code & 0x7F,
354        ])
355    if isinstance(msg, CabBusUnknown):
356        return bytes([0x80 | msg.address]) + msg.data
357    raise NotImplementedError(f"encode not implemented for {type(msg).__name__}")

219        return FastClockUpdate(address=addr, hour=body[0] & 0x7F, minute=body[1] & 0x7F)
220
221    if addr == ADDR_ACCESSORY and len(body) >= 3:
222        accy_addr = ((body[0] & 0x7F) << 7) | (body[1] & 0x7F)
223        closed = bool(body[2] & 0x01)
224        return AccessoryObserved(
225            address=addr, accessory_address=accy_addr + 1, closed=closed
226        )
227
228    if addr == ADDR_AIU_UPDATE and len(body) >= 3:
229        return AiuUpdate(
230            address=addr,
231            aiu_module=body[0] & 0x7F,
232            input_word=((body[1] & 0x7F) << 7) | (body[2] & 0x7F),
233        )
234
235    if addr == ADDR_THROTTLE_CMD and len(body) >= 4:
236        return ThrottleObserved(
237            address=addr,
238            dcc_address=((body[0] & 0x7F) << 7) | (body[1] & 0x7F),
239            speed=body[2] & 0x7F,
240            forward=bool(body[3] & 0x40),
241            functions=body[3] & 0x1F,
242        )
243
244    if addr == ADDR_PROG_RESP and len(body) >= 4:
245        return ProgrammingResponse(
246            address=addr,
247            cv=((body[0] & 0x7F) << 7) | (body[1] & 0x7F),
248            value=body[2] & 0xFF,
249            operation=body[3] & 0x03,
250        )
251
252    if addr == ADDR_CONSIST_CMD and len(body) >= 4:
253        return ConsistObserved(
254            address=addr,
255            consist_number=body[0] & 0x7F,
256            loco_address=((body[1] & 0x7F) << 7) | (body[2] & 0x7F),
257            operation=body[3] & 0x03,
258        )
259
260    if addr == ADDR_MACRO_CMD and len(body) >= 1:
261        return MacroObserved(
262            address=addr,
263            macro_number=body[0] & 0x7F,
264        )
265
266    if addr == ADDR_MASTER_ACK and len(body) >= 1:
267        return MasterAck(
268            address=addr,
269            code=body[0] & 0x7F,
270        )
271
272    return CabBusUnknown(address=addr, data=body)

frame starts with the address byte (high bit set). Everything after is body. Returns CabBusUnknown for unrecognized frames.

275def encode(msg: CabBusMessage) -> bytes:
276    """Inverse of :func:`decode` — used only by tests / synthetic playback.
277
278    The real monitor never writes to the bus (it's passive); we provide
279    ``encode`` to construct test fixtures that feed the monitor through a
280    loopback transport.
281    """
282    if isinstance(msg, CabPoll):
283        return bytes([0x80 | msg.address, msg.command])
284    if isinstance(msg, CabResponse):
285        # 9-byte body: 5 state bytes + 3 pad + 1 checksum. The
286        # checksum is the XOR of the 5 state bytes on the wire — we
287        # emit it here so encode() → decode() round-trips against
288        # the tightened 9-byte length check in ``decode``.
289        state = (
290            msg.key & 0x7F,
291            msg.speed & 0x7F,
292            msg.direction & 0x7F,
293            msg.function_low & 0x7F,
294            msg.function_high & 0x7F,
295        )
296        checksum = 0
297        for b in state:
298            checksum ^= b
299        return bytes([
300            0x80 | msg.address,
301            *state,
302            0x00, 0x00, 0x00,
303            checksum & 0x7F,
304        ])
305    if isinstance(msg, FastClockUpdate):
306        return bytes([0x80 | ADDR_FAST_CLOCK, msg.hour & 0x7F, msg.minute & 0x7F])
307    if isinstance(msg, AccessoryObserved):
308        a = msg.accessory_address - 1
309        return bytes([
310            0x80 | ADDR_ACCESSORY,
311            (a >> 7) & 0x7F,
312            a & 0x7F,
313            0x01 if msg.closed else 0x00,
314        ])
315    if isinstance(msg, AiuUpdate):
316        return bytes([
317            0x80 | ADDR_AIU_UPDATE,
318            msg.aiu_module & 0x7F,
319            (msg.input_word >> 7) & 0x7F,
320            msg.input_word & 0x7F,
321        ])
322    if isinstance(msg, ThrottleObserved):
323        return bytes([
324            0x80 | ADDR_THROTTLE_CMD,
325            (msg.dcc_address >> 7) & 0x7F,
326            msg.dcc_address & 0x7F,
327            msg.speed & 0x7F,
328            (msg.functions & 0x1F) | (0x40 if msg.forward else 0),
329        ])
330    if isinstance(msg, ProgrammingResponse):
331        return bytes([
332            0x80 | ADDR_PROG_RESP,
333            (msg.cv >> 7) & 0x7F,
334            msg.cv & 0x7F,
335            msg.value & 0x7F,
336            msg.operation & 0x7F,
337        ])
338    if isinstance(msg, ConsistObserved):
339        return bytes([
340            0x80 | ADDR_CONSIST_CMD,
341            msg.consist_number & 0x7F,
342            (msg.loco_address >> 7) & 0x7F,
343            msg.loco_address & 0x7F,
344            msg.operation & 0x7F,
345        ])
346    if isinstance(msg, MacroObserved):
347        return bytes([
348            0x80 | ADDR_MACRO_CMD,
349            msg.macro_number & 0x7F,
350        ])
351    if isinstance(msg, MasterAck):
352        return bytes([
353            0x80 | ADDR_MASTER_ACK,
354            msg.code & 0x7F,
355        ])
356    if isinstance(msg, CabBusUnknown):
357        return bytes([0x80 | msg.address]) + msg.data
358    raise NotImplementedError(f"encode not implemented for {type(msg).__name__}")
